import catch_up.FourCounterEndgame

/-!
# Catch-Up four-counter endgame executable

Test tied top-four endgames through `N = 100` with:

```bash
lake env lean --run scripts/catch_up/FourCounterEndgameMain.lean
```

Pass another nonnegative bound as the sole argument to enlarge or shrink the tied test.
-/

namespace CatchUpSearch.FourCounterEndgame

open CatchUp

def outcomeSymbol : Outcome → String
  | .win => "W"
  | .draw => "D"
  | .loss => "L"

def formatProfile (profile : List Outcome) : String :=
  String.intercalate "" <| profile.map outcomeSymbol

def parseBound (argument : String) : IO ℕ :=
  match argument.toNat? with
  | some bound => pure bound
  | none => throw <| IO.userError s!"Expected a natural-number bound, got '{argument}'."

def orThrow {α : Type} (result : Except String α) : IO α :=
  match result with
  | .error message => throw <| IO.userError message
  | .ok value => pure value

def run (bound : ℕ) : IO Unit := do
  IO.println s!"tied failures through N={bound}: {tiedFailuresThrough bound}"
  let checkedFailures ← orThrow (checkedDrawFailuresThrough bound)
  IO.println s!"checked-certificate failures through N={bound}: {checkedFailures}"
  IO.println "nearby deficit profiles (index = deficit; W/D/L = current mover outcome):"
  for N in List.range 13 do
    IO.println s!"N={N}: {formatProfile (deficitProfile N (N + 8))}"
  IO.println "empty tied state reachable after opening 1 at parity-eligible sizes through N=12:"
  for N in [3, 4, 7, 8, 11, 12] do
    let reachable ← orThrow (tiedEmptyReachableAfterOpeningOne N)
    IO.println s!"N={N}: {reachable}"

def runArgs (args : List String) : IO Unit :=
  match args with
  | [] => run 100
  | [argument] => do
      run (← parseBound argument)
  | _ => throw <| IO.userError "Usage: FourCounterEndgameMain.lean [bound]"

end CatchUpSearch.FourCounterEndgame

set_option linter.style.namespace false in
def main (args : List String) : IO Unit :=
  CatchUpSearch.FourCounterEndgame.runArgs args
