import catch_up.FourCounterBoundary

/-!
# Catch-Up four-counter forcing-basin boundary executable

Inspect the failed safe-top-only forcing basins and test both the semantic one-old enlargement and
the explicit three-counter target family at `N = 8` and `N = 11` with:

```bash
lake env lean --run scripts/catch_up/FourCounterBoundaryMain.lean
```

Pass small game sizes to select other `N → N + 4` pairs whose opening-`1` roots lie outside the
basin.
-/

namespace CatchUpSearch.FourCounterBoundary

open CatchUp
open CatchUpSearch.ExactSolver

def playerName : Player → String
  | .p1 => "p1"
  | .p2 => "p2"

def formatCounters (counters : List ℕ) : String :=
  "[" ++ String.intercalate "," (counters.map toString) ++ "]"

def oldCounterValues (N : ℕ) (state : State) : List ℕ :=
  (availableMoves N (PairedTrace.oldMask N state)).map fun move => move.2

def topCounterValues (N : ℕ) (state : State) : List ℕ :=
  ((availableMoves (N + 4) state.remainingMask).filter fun move => decide (N < move.2)).map
    fun move => move.2

def formatMove (N : ℕ) (move : MoveEvaluation) : String :=
  s!"{move.counter} → mover={playerName move.child.toMove}, deficit={move.child.deficit}, " ++
    s!"old={formatCounters (oldCounterValues N move.child)}, " ++
    s!"top={formatCounters (topCounterValues N move.child)}"

def formatMoves (N : ℕ) (moves : List MoveEvaluation) : String :=
  String.intercalate "; " (moves.map (formatMove N))

def BoundaryWitness.format (N : ℕ) (witness : BoundaryWitness) : String :=
  s!"depth={witness.depth}; mover={playerName witness.state.toMove}; " ++
    s!"deficit={witness.state.deficit}; " ++
    s!"old={formatCounters (oldCounterValues N witness.state)}; " ++
    s!"top={formatCounters (topCounterValues N witness.state)}\n" ++
    s!"    enters: {formatMoves N witness.enters}\n" ++
    s!"    escapes: {formatMoves N witness.escapes}"

def formatWitnesses (N : ℕ) (label : String) (witnesses : List BoundaryWitness) : String :=
  if witnesses.isEmpty then s!"  no {label} boundary witnesses"
  else
    String.intercalate "\n" <| witnesses.map fun witness =>
      s!"  {label} boundary:\n  " ++ witness.format N

def formatHistogram (histogram : List (ℕ × ℕ)) : String :=
  String.intercalate ", " <| histogram.map fun entry => s!"{entry.1}:{entry.2}"

def BoundarySummary.format (summary : BoundarySummary) : String :=
  let root := match summary.rootBoundary with
    | none => "  root is not a mixed opponent boundary"
    | some witness => "  root boundary:\n  " ++ witness.format summary.smallN
  let earliest := formatWitnesses summary.smallN "earliest" summary.earliestWitnesses
  let positive :=
    formatWitnesses summary.smallN "closest-positive-old" summary.closestPositiveOldWitnesses
  let topOnly := formatWitnesses summary.smallN "top-only" summary.topOnlyWitnesses
  let explicitRoot := match summary.explicitRootBoundary with
    | none => "  root is not a mixed boundary for the explicit target"
    | some witness => "  explicit-target root boundary:\n  " ++ witness.format summary.smallN
  let explicitOneOld :=
    formatWitnesses summary.smallN "explicit-target-one-old"
      summary.explicitClosestOneOldWitnesses
  s!"N={summary.smallN} → {summary.smallN + 4}; owner={playerName summary.owner}; " ++
    s!"graph={summary.graphNodes}; basin={summary.basinNodes}; " ++
    s!"one-old-target={summary.oneOldTargetNodes}; " ++
    s!"one-old-basin={summary.oneOldBasinNodes}; " ++
    s!"one-old-root={summary.rootInOneOldBasin}; " ++
    s!"explicit-target={summary.explicitTargetNodes}; " ++
    s!"explicit-basin={summary.explicitBasinNodes}; " ++
    s!"explicit-root={summary.rootInExplicitBasin}; " ++
    s!"explicit-reachable-safe-outside={summary.explicitReachableSafeOutsideNodes}; " ++
    s!"explicit-reachable-safe-boundary={summary.explicitReachableSafeBoundaryNodes}; " ++
    s!"reachable-safe-outside={summary.reachableSafeOutsideNodes}; " ++
    s!"global-safe-boundary={summary.globalSafeBoundaryNodes}; " ++
    s!"reachable-safe-boundary={summary.reachableSafeBoundaryNodes}; " ++
    s!"terminal-safe-outside={summary.terminalSafeOutsideNodes}; " ++
    s!"min-boundary-old={summary.minimumBoundaryOldCounters}; " ++
    s!"min-positive-boundary-old={summary.minimumPositiveBoundaryOldCounters}\n" ++
    s!"  entering counters: {formatHistogram summary.enteringCounterHistogram}\n" ++
    s!"  escaping counters: {formatHistogram summary.escapingCounterHistogram}\n" ++
    root ++ "\n" ++ earliest ++ "\n" ++ positive ++ "\n" ++ topOnly ++ "\n" ++
    explicitRoot ++ "\n" ++ explicitOneOld

def orThrow {α : Type} (result : Except String α) : IO α :=
  match result with
  | .error message => throw <| IO.userError message
  | .ok value => pure value

def parseSize (argument : String) : IO ℕ :=
  match argument.toNat? with
  | some N => pure N
  | none => throw <| IO.userError s!"Expected a natural-number game size, got '{argument}'."

def runSize (N : ℕ) : IO Unit := do
  IO.eprintln s!"solving N={N} → {N + 4} opening-1 graph..."
  let output ← orThrow (PairedTrace.solveOpeningGraph (N + 4) 1)
  IO.eprintln s!"solved N={N} → {N + 4}: {output.nodes.length} graph nodes"
  for owner in [.p1, .p2] do
    IO.eprintln s!"inspecting safe-top-only boundary for owner={playerName owner}..."
    IO.println (← orThrow (analyzeOutput N owner output)).format

def runArgs (args : List String) : IO Unit := do
  let sizes ← if args.isEmpty then pure [8, 11] else args.mapM parseSize
  for N in sizes do
    runSize N

end CatchUpSearch.FourCounterBoundary

set_option linter.style.namespace false in
def main (args : List String) : IO Unit :=
  CatchUpSearch.FourCounterBoundary.runArgs args
