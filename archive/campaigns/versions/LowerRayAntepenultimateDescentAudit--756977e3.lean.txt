import catch_up.LowerRayAntepenultimateDescent

/-!
# Antepenultimate endpoint statement and axiom audit

This audit restates the endpoint transfer using set difference and rejects any
axiom outside the standard three. Run the module with `lake --wfail build`.
-/

namespace CatchUp.EndpointTransferAudit

/-- The endpoint transfer in the requested set-difference notation. -/
@[category test, AMS 91]
theorem set_difference_form (N : ℕ) (hN : 7 ≤ N)
    (hdraw : (afterOpening (Finset.Icc 1 (N - 4)) 1).value = .draw) :
    Position.value ⟨Finset.Icc 1 N \ {1, N - 2}, N - 3⟩ ≠ .loss := by
  have hremaining : Finset.Icc 1 N \ {1, N - 2} =
      ((Finset.Icc 1 N).erase 1).erase (N - 2) := by
    ext x
    simp only [Finset.mem_sdiff, Finset.mem_insert, Finset.mem_singleton,
      Finset.mem_erase]
    tauto
  simpa only [hremaining, pairDeletionPosition, Nat.sub_sub] using
    pair_deletion_position_antepenultimate_value_ne_loss_of_opening_one_draw N hN hdraw

open Lean Elab Command in
run_cmd do
  let declarations := #[
    ``pair_deletion_position_antepenultimate_value_ne_loss_of_opening_one_draw,
    ``set_difference_form]
  let allowed := #[``propext, ``Classical.choice, ``Quot.sound]
  for declaration in declarations do
    let actual ← Lean.collectAxioms declaration
    for axiomName in actual do
      unless allowed.contains axiomName do
        throwError "Unexpected axiom in {declaration}: {axiomName}"
    logInfo m!"Verified axioms for {declaration}: {actual}"

end CatchUp.EndpointTransferAudit
