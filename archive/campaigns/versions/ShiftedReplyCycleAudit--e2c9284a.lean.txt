import catch_up.ShiftedReplyCycle

/-!
# Transitive axiom audit for shifted reply cycle results

Every public theorem in `ShiftedReplyCycle` is checked against the standard logical
axiom whitelist. Conjecture placeholders and native-evaluation axioms are excluded.
-/

open Lean Elab Command in
run_cmd do
  let declarations := #[
    ``CatchUp.ShiftedReplyCycle.edge_irrefl,
    ``CatchUp.ShiftedReplyCycle.adjacent_forward,
    ``CatchUp.ShiftedReplyCycle.adjacent_reverse,
    ``CatchUp.ShiftedReplyCycle.nonadjacent_reversal,
    ``CatchUp.ShiftedReplyCycle.obstruction_tail_losses,
    ``CatchUp.ShiftedReplyCycle.obstruction_structure,
    ``CatchUp.ShiftedReplyCycle.obstruction_two_cycle,
    ``CatchUp.ShiftedReplyCycle.obstruction_root_loss,
    ``CatchUp.ShiftedReplyCycle.no_local_span_two_cycle_exclusion,
    ``CatchUp.ShiftedReplyCycle.no_loss_conditioned_local_span_exclusion,
    ``CatchUp.ShiftedReplyCycle.interior_structure,
    ``CatchUp.ShiftedReplyCycle.interior_slide_failure,
    ``CatchUp.ShiftedReplyCycle.no_full_span_minimum_slide,
    ``CatchUp.ShiftedReplyCycle.sparse_three_cycle,
    ``CatchUp.ShiftedReplyCycle.interval_seven_upward_edge,
    ``CatchUp.ShiftedReplyCycle.interval_seven_max_edge,
    ``CatchUp.ShiftedReplyCycle.interval_seven_no_max_companion,
    ``CatchUp.ShiftedReplyCycle.win_has_move,
    ``CatchUp.ShiftedReplyCycle.loss_forces_outgoing,
    ``CatchUp.ShiftedReplyCycle.finite_outgoing_closed_walk,
    ``CatchUp.ShiftedReplyCycle.loss_forces_closed_walk,
    ``CatchUp.ShiftedReplyCycle.interval_loss_forces_closed_walk,
    ``CatchUp.ShiftedReplyCycle.critical_cross,
    ``CatchUp.ShiftedReplyCycle.critical_reply_inversion,
    ``CatchUp.ShiftedReplyCycle.interval_seven_critical_continuation]
  let allowed := #[``propext, ``Classical.choice, ``Quot.sound]
  for declaration in declarations do
    let actual ← Lean.collectAxioms declaration
    for axiomName in actual do
      unless allowed.contains axiomName do
        throwError "Unexpected axiom in {declaration}: {axiomName}"
    logInfo m!"Verified axioms for {declaration}: {actual}"
