import catch_up.PairedTrace

/-!
# Catch-Up paired normalized strategy-trace executable

Summarize the default discovery pairs `7 → 11` and `8 → 12` with:

```bash
lake env lean --run scripts/catch_up/PairedTraceMain.lean
```

Pass small game sizes to choose other pairs:

```bash
lake env lean --run scripts/catch_up/PairedTraceMain.lean 11 12
```

Export every row for one pair as tab-separated data with:

```bash
lake env lean --run scripts/catch_up/PairedTraceMain.lean --tsv 7
```
-/

namespace CatchUpSearch.PairedTrace

def parseGameSize (argument : String) : IO ℕ :=
  match argument.toNat? with
  | some N => pure N
  | none => throw <| IO.userError s!"Expected a natural-number game size, got '{argument}'."

def runArgs (args : List String) : IO Unit :=
  match args with
  | [] => runSummaries [7, 8]
  | ["--tsv", argument] => do
      runTsv (← parseGameSize argument)
  | "--tsv" :: _ => throw <| IO.userError "Usage: PairedTraceMain.lean --tsv N"
  | arguments => do
      runSummaries (← arguments.mapM parseGameSize)

end CatchUpSearch.PairedTrace

set_option linter.style.namespace false in
def main (args : List String) : IO Unit :=
  CatchUpSearch.PairedTrace.runArgs args
