import catch_up.AdjacentHoleExchangeObstruction
import catch_up.DeficitWinningBound

/-!
# Axiom audit for the adjacent-hole exchange obstructions

Reject every axiom outside the standard three. The imported opening-polarity API
is audited alongside every public theorem added in the obstruction module.
-/

namespace CatchUp.AdjacentHoleExchange

open Lean Elab Command in
run_cmd do
  let declarations := #[
    ``left_value, ``right_value,
    ``left_move_profile, ``right_move_profile,
    ``safe_right_move_copies_to_loss,
    ``left_eq_requested, ``right_eq_requested,
    ``witness_admissible, ``right_to_left_counter_legal,
    ``forced_first_repair, ``forced_first_repair_endpoints,
    ``alternating_parity_jump, ``alternating_witness_structure,
    ``parity_root_opening_five_draw, ``parity_root_pair_erasure,
    ``omitted_pairs_geometry, ``parity_root_neighbor_states,
    ``parity_root_reply_eleven_draw,
    ``CatchUp.after_opening_loss_forces_neighbor_value_jump]
  let allowed := #[``propext, ``Classical.choice, ``Quot.sound]
  for declaration in declarations do
    let actual ← Lean.collectAxioms declaration
    for axiomName in actual do
      unless allowed.contains axiomName do
        throwError "Unexpected axiom in {declaration}: {axiomName}"
    logInfo m!"Verified axioms for {declaration}: {actual}"

end CatchUp.AdjacentHoleExchange
