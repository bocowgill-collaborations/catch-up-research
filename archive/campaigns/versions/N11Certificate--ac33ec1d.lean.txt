/-!
# A formally checked Catch-Up draw certificate for eleven counters

This module contains deterministic one-sided partial-strategy DAGs for the interval game on
`{1, ..., 11}`. The player-1 DAG covers opening `1`; the player-2 DAG covers every legal opening.
Target leaves use the exact proof-backed six-counter safety classifier. The independent checker and
its soundness theorem turn the literal data into a proof of the complete game's draw value.
-/

/-- The proof-backed target used by the player-1 certificate. -/
def playerOneTarget : TargetPredicate :=
  smallSafeTarget 6 playerOneDAG.gameSize playerOneDAG.player

/-- The proof-backed target used by the player-2 certificate. -/
def playerTwoTarget : TargetPredicate :=
  smallSafeTarget 6 playerTwoDAG.gameSize playerTwoDAG.player

/-- The literal player-1 partial certificate passes the independent checker. -/
@[category test, AMS 91]
theorem player_one_certificate_checks :
    checkPartialCertificate playerOneTarget playerOneDAG = true := by
  native_decide

/-- The literal player-2 all-openings partial certificate passes the independent checker. -/
@[category test, AMS 91]
theorem player_two_certificate_checks :
    checkPartialCertificate playerTwoTarget playerTwoDAG = true := by
  native_decide

/-- The encoded post-opening state for each legal opening of the eleven-counter game. -/
def openingState : ℕ → State
  | 1 => ⟨2046, 1, .p2⟩
  | 2 => ⟨2045, 2, .p2⟩
  | 3 => ⟨2043, 3, .p2⟩
  | 4 => ⟨2039, 4, .p2⟩
  | 5 => ⟨2031, 5, .p2⟩
  | 6 => ⟨2015, 6, .p2⟩
  | 7 => ⟨1983, 7, .p2⟩
  | 8 => ⟨1919, 8, .p2⟩
  | 9 => ⟨1791, 9, .p2⟩
  | 10 => ⟨1535, 10, .p2⟩
  | 11 => ⟨1023, 11, .p2⟩
  | _ => ⟨0, 0, .p2⟩

/-- Every legal encoded opening state decodes to the formal post-opening game position. -/
@[category test, AMS 91]
theorem to_game_position_opening_state (x : ℕ) (hx : x ∈ Finset.Icc 1 11) :
    toGamePosition 11 (openingState x) =
      GamePosition.afterOpening (Finset.Icc 1 11) x := by
  have hxLower : 1 ≤ x := (Finset.mem_Icc.mp hx).1
  have hxUpper : x ≤ 11 := (Finset.mem_Icc.mp hx).2
  interval_cases x <;> decide

/-- The player-2 DAG contains the encoded state after every legal opening. -/
@[category test, AMS 91]
theorem player_two_contains_opening_state (x : ℕ) (hx : x ∈ Finset.Icc 1 11) :
    ContainsState playerTwoDAG (openingState x) := by
  have hxLower : 1 ≤ x := (Finset.mem_Icc.mp hx).1
  have hxUpper : x ≤ 11 := (Finset.mem_Icc.mp hx).2
  interval_cases x <;> decide

/-- Player 1 can avoid losing after opening with counter one. -/
@[category test, AMS 91]
theorem player_one_can_avoid_loss :
    CanAvoidLoss .p1 (GamePosition.afterOpening (Finset.Icc 1 11) 1) := by
  have htarget : TargetSound playerOneTarget playerOneDAG := by
    simpa only [playerOneTarget] using small_safe_target_sound 6 playerOneDAG
  have hsafe := check_partial_certificate_sound playerOneTarget playerOneDAG htarget
    player_one_certificate_checks
  simpa [playerOneDAG, toGamePosition, GamePosition.afterOpening, CatchUp.afterOpening] using hsafe

/-- Player 2 can avoid losing after every legal opening. -/
@[category test, AMS 91]
theorem player_two_can_avoid_loss (x : ℕ) (hx : x ∈ Finset.Icc 1 11) :
    CanAvoidLoss .p2 (GamePosition.afterOpening (Finset.Icc 1 11) x) := by
  have htarget : TargetSound playerTwoTarget playerTwoDAG := by
    simpa only [playerTwoTarget] using small_safe_target_sound 6 playerTwoDAG
  have hvalid : PartialDAGValid playerTwoTarget playerTwoDAG :=
    of_decide_eq_true (by
      simpa only [checkPartialCertificate] using player_two_certificate_checks)
  have hsafe := can_avoid_loss_of_contains_state_partial playerTwoTarget playerTwoDAG htarget
    hvalid (openingState x) (player_two_contains_opening_state x hx)
  change CanAvoidLoss .p2 (toGamePosition 11 (openingState x)) at hsafe
  rw [to_game_position_opening_state x hx] at hsafe
  exact hsafe

/-- Optimal play in Catch-Up on `{1, ..., 11}` results in a draw. -/
@[category test, AMS 91]
theorem value_Icc_eleven :
    value (Finset.Icc 1 11) = .draw := by
  apply (value_eq_draw_iff_can_avoid_loss_after_opening (Finset.Icc 1 11)
    (Finset.nonempty_Icc.mpr (by decide))).2
  constructor
  · exact ⟨1, by decide, player_one_can_avoid_loss⟩
  · exact player_two_can_avoid_loss

end CatchUpSearch.N11Certificate
