#!/usr/bin/env python3
"""Bounded first-pair diagnostics, not a strategy invariant or an all-N proof.

Counter x is bit x-1. The imported exact completed-margin recurrence removes
one counter per step; equality switches the mover. Independent absolute-score
play does not use the early-terminal cutoff. No controller table is extended.
"""

import argparse
import json
from pathlib import Path
import sys

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "scripts" / "catch_up"))
from two_buffer_controller import absolute_margin, counters, margin, mask, total


def move_margin(xs, deficit, x):
    assert x in xs
    bits = mask(xs) ^ (1 << (x - 1))
    if x < deficit:
        return margin(bits, deficit - x)
    return -margin(bits, x - deficit)


def checked_profile(xs, deficit):
    xs = tuple(sorted(xs))
    normalized = margin(mask(xs), deficit)
    independent = absolute_margin(xs, 0, deficit, 0)
    assert normalized == independent
    moves = {}
    for x in xs:
        nxt = tuple(y for y in xs if y != x)
        nxt_mover = 0 if x < deficit else 1
        independent_child = absolute_margin(nxt, x, deficit, nxt_mover)
        result = move_margin(xs, deficit, x)
        assert result == independent_child
        moves[str(x)] = result
    return {"remaining": xs, "deficit": deficit,
            "completed_margin": normalized, "moves": moves}


def absolute_trace(xs, deficit, picks):
    remaining = set(xs)
    scores = [0, deficit]
    mover = 0
    events = []
    for x in picks:
        assert remaining and sum(remaining) >= scores[1-mover] - scores[mover]
        assert x in remaining
        old_mover = mover
        remaining.remove(x)
        scores[mover] += x
        if scores[mover] >= scores[1-mover]:
            mover = 1-mover
        events.append({"pick": x, "owner": old_mover,
                       "scores": scores.copy(), "next_mover": mover,
                       "remaining": sorted(remaining)})
    return {"events": events, "remaining": sorted(remaining), "scores": scores,
            "mover": mover, "deficit": scores[1-mover] - scores[mover]}


def run():
    # Direct endpoint evaluations, not deductions from tied interval-root draws.
    endpoints = [checked_profile(range(3, n+1), 1) for n in (7, 8, 11, 12)]
    assert all(p["completed_margin"] == 0 for p in endpoints)

    sparse = (3, 7, 9, 14, 17, 20, 29, 34)
    actual = checked_profile(sparse, 1)
    shadow = checked_profile(sparse[1:], 2)
    assert (1 + sum(sparse)) % 2 == 0 and min(sparse) == 3
    assert actual["completed_margin"] == -2
    assert shadow["completed_margin"] == 2
    assert all(v < 0 for v in actual["moves"].values())
    children = [{"pick": x, "next_mover_margin": margin(mask(sparse) ^ (1 << (x-1)), x-1)}
                for x in sparse]
    assert all(c["next_mover_margin"] > 0 for c in children)

    actual_bridge = checked_profile((3, 4, 6, 8), 3)
    shadow_bridge = checked_profile((4, 6, 8), 4)
    assert actual_bridge["completed_margin"] == shadow_bridge["completed_margin"] == 2
    assert [int(x) for x, v in shadow_bridge["moves"].items() if v > 0] == [8]
    assert actual_bridge["moves"]["8"] == -2
    assert actual_bridge["moves"]["6"] == 2
    actual_prefix = absolute_trace(range(3, 9), 1, (5, 7))
    shadow_prefix = absolute_trace(range(4, 9), 2, (5, 7))
    assert (actual_prefix["remaining"], actual_prefix["deficit"], actual_prefix["mover"]) == (
        [3, 4, 6, 8], 3, 0)
    assert (shadow_prefix["remaining"], shadow_prefix["deficit"], shadow_prefix["mover"]) == (
        [4, 6, 8], 4, 0)
    # This limitation is essential: the shadow does not win from this interval root.
    assert margin(mask(range(4, 9)), 2) == 0
    after_copy = [checked_profile((3, 4, 6), 5), checked_profile((4, 6), 4)]
    assert [p["completed_margin"] for p in after_copy] == [2, -2]
    actual_copy = absolute_trace((3, 4, 6, 8), 3, (8, 4, 6))
    shadow_copy = absolute_trace((4, 6, 8), 4, (8, 4, 6))
    assert [e["owner"] for e in actual_copy["events"]] == [0, 1, 1]
    assert [e["owner"] for e in shadow_copy["events"]] == [0, 1, 0]
    assert actual_copy["remaining"] == [3] and actual_copy["deficit"] == 5
    assert shadow_copy["remaining"] == [] and shadow_copy["deficit"] == 6
    actual_completion = actual_copy["scores"].copy()
    actual_completion[actual_copy["mover"]] += sum(actual_copy["remaining"])
    actual_margin = actual_completion[0] - actual_completion[1]
    shadow_margin = shadow_copy["scores"][0] - shadow_copy["scores"][1]
    assert (actual_margin, shadow_margin) == (-2, 6)

    # Reproduce the failed greedy continuation after a safe first pick three.
    greedy = absolute_trace(range(3, 8), 1, (3, 7, 4, 5, 6))
    assert greedy["remaining"] == [] and greedy["scores"] == [12, 14]

    # Small falsification probe of the stronger interval-at-deficit-one statement.
    interval_tests = []
    for a in range(1, 31):
        for length in range(1, 13):
            b = a + length - 1
            xs = tuple(range(a, b+1))
            if sum(xs) % 2:
                value = margin(mask(xs), 1)
                interval_tests.append([a, b, value])
    assert len(interval_tests) == 180
    assert all(v >= 0 for _, _, v in interval_tests)

    # Broader small-deficit transfer is false; this is not a min-three witness.
    tied_support = (1, 6, 11, 14, 15, 18, 23)
    tied = checked_profile(tied_support, 0)
    two = checked_profile(tied_support, 2)
    assert (tied["completed_margin"], two["completed_margin"]) == (2, -2)

    return {"format": "catchup-first-pair-diagnostics-v1",
            "trust": "Exact bounded computation; Lean separately certifies the stated outcome claims.",
            "endpoints": endpoints, "sparse_actual": actual, "sparse_shadow": shadow,
            "sparse_all_children": children,
            "bridge_actual": actual_bridge, "bridge_shadow": shadow_bridge,
            "actual_prefix": actual_prefix, "shadow_prefix": shadow_prefix,
            "copied_eight_children": after_copy,
            "actual_copied_turns": actual_copy, "shadow_copied_turns": shadow_copy,
            "completed_copied_margins": [actual_margin, shadow_margin],
            "bridge_limitation": "Actual bridge wins; shadow interval root draws. Not a refutation of loss-conditioned copying.",
            "greedy_counterplay": greedy,
            "interval_probe": interval_tests,
            "general_tied_to_deficit_two_counterexample": [tied, two]}


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path)
    args = parser.parse_args()
    results = run()
    encoded = json.dumps(results, indent=2, sort_keys=True) + "\n"
    if args.output:
        args.output.write_text(encoded)
    print(json.dumps({"checks": "PASS", "endpoint_sizes": [7, 8, 11, 12],
                      "sparse_margins": [-2, 2], "interval_tests": 180,
                      "output": str(args.output) if args.output else None}))
