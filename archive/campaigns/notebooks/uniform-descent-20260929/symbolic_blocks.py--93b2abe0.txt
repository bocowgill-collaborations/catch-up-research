"""Bounded symbolic exact-sum proof search, NOT an outcome solver.

Large counters have the form m+q; small counters are positive constants >=2.
The support contains [2,m] and selected positive offsets, minus the recorded
small and large deletions. A node of deficit m+t selects one large counter m+q
and one or two unused small counters summing t-q. ExactSumLossObstruction then
requires all still available offsets w>q to have nonloss proofs. All arithmetic
identities are independent of m once small/large separation is imposed.

Search failure is only failure of the bounded grammar. A returned tree is not
a Lean proof; it must be replayed and translated with explicit bounds on m.
"""

import argparse
import json
import time
from functools import cache


def search(k, opening, depth, radius):
    top = frozenset(range(3, k + 3))
    nodes = 0

    @cache
    def visit(t, small, deleted, fuel):
        nonlocal nodes
        nodes += 1
        if fuel == 0:
            return None
        available = [q for q in range(-radius, k + 3)
                     if (q <= 0 or q in top) and q not in deleted]
        for q in reversed(available):
            r = t - q
            if not 2 <= r <= 2 * radius:
                continue
            blocks = []
            if r not in small:
                blocks.append((r,))
            blocks += [(a, r - a) for a in range(2, (r + 1) // 2)
                       if a not in small and r - a not in small]
            for block in blocks:
                new_small = tuple(sorted((*small, *block)))
                replies = [w for w in available if w > q]
                children = []
                for w in replies:
                    child = visit(w, new_small, tuple(sorted((*deleted, q, w))), fuel - 1)
                    if child is None:
                        break
                    children.append((w, child))
                else:
                    return {"deficit_offset": t, "pivot_offset": q,
                            "small_block": block, "children": children}
        return None

    tree = visit(opening, (), (opening,), depth)
    return {"k": k, "opening_offset": opening, "depth_bound": depth,
            "radius": radius, "visited": nodes, "tree": tree}


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--k", type=int, default=4)
    parser.add_argument("--depth", type=int, default=6)
    parser.add_argument("--radius", type=int, default=8)
    args = parser.parse_args()
    started = time.monotonic()
    for opening in range(3, args.k + 3):
        print(json.dumps(search(args.k, opening, args.depth, args.radius)), flush=True)
    print(json.dumps({"elapsed_seconds": time.monotonic() - started}))
