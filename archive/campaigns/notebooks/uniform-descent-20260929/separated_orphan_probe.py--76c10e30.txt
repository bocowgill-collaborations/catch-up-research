"""Falsify the proposed extension of the PROVED one-pair orphan repair.

Replace one adjacent pair by three disjoint adjacent pairs, each starting at
least max(a+2,2a-1). Test nonwinning at deficits a-1 and a+1. The one-pair
case has an independent parametric proof; this extension does not.
"""
import itertools
import json
import sys
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2] / "scripts/catch_up"))
from adjacent_jump_probe import mask_of, state, value, verify_independently

started = time.monotonic()
tested = 0
for a in range(2,11):
    low = max(a+2,2*a-1)
    for starts in itertools.combinations(range(low,low+12),3):
        if any(y<=x+1 for x,y in zip(starts,starts[1:])):
            continue
        mask = mask_of([a,*[x for b in starts for x in (b,b+1)]])
        for d in (a-1,a+1):
            tested += 1
            if value(mask,d)==1:
                verify_independently(mask,d)
                print(json.dumps({"refuted":True,"a":a,"pair_starts":starts,
                                  "state":state(mask,d),"tested":tested,
                                  "elapsed":time.monotonic()-started}))
                raise SystemExit(0)
        value.cache_clear()
print(json.dumps({"refuted":False,"tested":tested,
                  "elapsed":time.monotonic()-started}))
