import catch_up.DenseThreeHoleFiniteChecker
import catch_up.PartialCertificateSoundness

/-!
# Shared strategy certificates for the dense three-hole invariant

This module extends the independent partial-strategy checker with a finite coverage condition for
every root quantified by `DenseThreeHoleNonloss`. A single checked strategy DAG may therefore
share descendants among all qualifying roots. The generator and exact solver are deliberately not
imported here.
-/

namespace CatchUpSearch.DenseThreeHoleCertificateChecker

open CatchUp
open CertificateChecker

/-- Check that every bounded support and deficit satisfying the dense hypotheses is represented
as a formal position in the supplied strategy DAG. -/
def coverageCheck (N : ℕ) (dag : StrategyDAG) : Bool :=
  (Finset.Icc 1 N).powerset.attach.fold (· && ·) true fun
      remaining : {s // s ∈ (Finset.Icc 1 N).powerset} =>
    (Finset.range (N + 1)).attach.fold (· && ·) true fun
        deficit : {d // d ∈ Finset.range (N + 1)} =>
      !DenseThreeHoleFiniteChecker.qualifies N remaining.1 deficit.1 ||
        decide (ContainsPosition dag ⟨⟨remaining.1, deficit.1⟩, .p1⟩)

/-- Independently check metadata, every partial-strategy obligation, and dense-root coverage. -/
def checkCertificate (N : ℕ) (target : TargetPredicate) (dag : StrategyDAG) : Bool :=
  decide (dag.gameSize = N ∧ dag.player = .p1) &&
    (checkPartialCertificate target dag && coverageCheck N dag)

/-- A checked shared strategy DAG with a sound target predicate proves the full dense three-hole
nonloss invariant. -/
@[category API, AMS 91]
theorem dense_three_hole_nonloss_of_check_certificate
    (N : ℕ) (target : TargetPredicate) (dag : StrategyDAG)
    (htarget : TargetSound target dag)
    (hcheck : checkCertificate N target dag = true) :
    DenseThreeHoleNonloss N := by
  have hparts :
      decide (dag.gameSize = N ∧ dag.player = .p1) = true ∧
        (checkPartialCertificate target dag = true ∧ coverageCheck N dag = true) := by
    simpa [checkCertificate, Bool.and_eq_true] using hcheck
  have hmeta : dag.gameSize = N ∧ dag.player = .p1 := of_decide_eq_true hparts.1
  have hvalid : PartialDAGValid target dag :=
    of_decide_eq_true (by simpa [checkPartialCertificate] using hparts.2.1)
  intro position hsub hholes hdN hcatch heven
  have hremaining : position.remaining ∈ (Finset.Icc 1 N).powerset := by
    simpa only [Finset.mem_powerset] using hsub
  have hdeficit : position.deficit ∈ Finset.range (N + 1) := by
    simp only [Finset.mem_range]
    omega
  have hcoverage := hparts.2.2
  rw [coverageCheck, Finset.fold_op_rel_iff_and
    (r := fun _ value : Bool => value = true) (c := true) (by simp)] at hcoverage
  simp only [true_and, Finset.mem_attach] at hcoverage
  have houter := hcoverage ⟨position.remaining, hremaining⟩ True.intro
  rw [Finset.fold_op_rel_iff_and
    (r := fun _ value : Bool => value = true) (c := true) (by simp)] at houter
  simp only [true_and, Finset.mem_attach] at houter
  have hinner := houter ⟨position.deficit, hdeficit⟩ True.intro
  have hqualifies :
      DenseThreeHoleFiniteChecker.qualifies N position.remaining position.deficit = true := by
    simp only [DenseThreeHoleFiniteChecker.qualifies, decide_eq_true_eq]
    exact ⟨hholes, hcatch, heven⟩
  have hcontains : ContainsPosition dag (⟨position, .p1⟩ : GamePosition) := by
    simpa [hqualifies] using hinner
  have hsafe := can_avoid_loss_of_contains_position_partial target dag htarget hvalid
    (⟨position, .p1⟩ : GamePosition) hcontains
  rw [can_avoid_loss_iff_value] at hsafe
  simpa [hmeta.2] using hsafe

end CatchUpSearch.DenseThreeHoleCertificateChecker
