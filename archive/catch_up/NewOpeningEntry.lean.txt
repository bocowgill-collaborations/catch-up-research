import catch_up.SimulationLedger

/-!
# New-opening entry phases for Catch-Up four-counter simulations

This module isolates a uniform opening phase in the comparison between Catch-Up games on
`Finset.Icc 1 N` and `Finset.Icc 1 (N + 4)`. When the smaller game opens at `k` and the larger
game opens at `N + k`, player two can mirror one complete turn from the smaller game and then
catch up in the larger game with a distinct added counter `N + j`. The resulting positions differ
by an explicit three-counter packet.

The results here prove only this parametric entry reduction. They do not assert that every
three-packet endpoint is safe.
-/

namespace CatchUp.NewOpeningEntry

/-- Bounds making `k` a small opening and `N + k` one of the four new openings. -/
def ValidNewOpening (N k : ℕ) : Prop :=
  1 ≤ k ∧ k ≤ 4 ∧ k ≤ N

/-- A distinct added counter that may end the large player's mirrored turn. -/
def ValidNewOpeningReply (k j : ℕ) : Prop :=
  1 ≤ j ∧ j ≤ 4 ∧ j ≠ k

/-- Counters present only in the large game during the mirrored part of the opening turn. -/
def newOpeningUnmatchedPacket (N k : ℕ) : Finset ℕ :=
  insert k (fourLiftTopCounters N \ {N + k})

/-- The three counters present only in the large game after the reply `N + j`. -/
def newOpeningThreePacketCounters (N k j : ℕ) : Finset ℕ :=
  insert k (fourLiftTopCounters N \ {N + k, N + j})

/-- A valid new-opening reply leaves exactly three unmatched counters. -/
theorem new_opening_three_packet_counters_card
    {N k j : ℕ} (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j) :
    (newOpeningThreePacketCounters N k j).card = 3 := by
  rcases hk with ⟨hkLower, hkUpper, hkN⟩
  rcases hj with ⟨hjLower, hjUpper, hjk⟩
  have hkTop : N + k ∈ fourLiftTopCounters N := by
    simp [fourLiftTopCounters]
    omega
  have hjTop : N + j ∈ fourLiftTopCounters N := by
    simp [fourLiftTopCounters]
    omega
  have hne : N + k ≠ N + j := by omega
  have hnotmem : k ∉ fourLiftTopCounters N \ {N + k, N + j} := by
    simp [fourLiftTopCounters]
    omega
  have hsubset : ({N + k, N + j} : Finset ℕ) ⊆ fourLiftTopCounters N := by
    intro counter hcounter
    simp only [Finset.mem_insert, Finset.mem_singleton] at hcounter
    rcases hcounter with rfl | rfl
    · exact hkTop
    · exact hjTop
  rw [newOpeningThreePacketCounters, Finset.card_insert_of_notMem hnotmem]
  rw [Finset.card_sdiff, Finset.inter_eq_left.mpr hsubset]
  rw [show ({N + k, N + j} : Finset ℕ).card = 2 by
    rw [Finset.card_insert_of_notMem]
    · simp
    · simpa using hne]
  simp [fourLiftTopCounters]

/-- Both games are in player two's turn, with the large deficit shifted upward by `N`. -/
structure NewOpeningMirrorPhase (N k : ℕ) (small large : GamePosition) : Prop where
  small_toMove : small.toMove = .p2
  large_toMove : large.toMove = .p2
  deficit_eq : large.deficit = N + small.deficit
  small_deficit_pos : 0 < small.deficit
  small_deficit_le : small.deficit ≤ k
  small_old : small.remaining ⊆ fourLiftOldCounters N
  k_not_small : k ∉ small.remaining
  remaining_eq : large.remaining = small.remaining ∪ newOpeningUnmatchedPacket N k

/-- The small game has caught up, while the mirrored large player still trails. -/
structure NewOpeningReadyPhase (N k : ℕ) (small large : GamePosition) : Prop where
  small_toMove : small.toMove = .p1
  large_toMove : large.toMove = .p2
  deficit_eq : large.deficit = N - small.deficit
  small_deficit_lt : small.deficit < N
  small_old : small.remaining ⊆ fourLiftOldCounters N
  k_not_small : k ∉ small.remaining
  remaining_eq : large.remaining = small.remaining ∪ newOpeningUnmatchedPacket N k

/-- Both games have passed to player one and differ by one explicit three-counter packet. -/
structure NewOpeningThreePacket (N k j : ℕ) (small large : GamePosition) : Prop where
  small_toMove : small.toMove = .p1
  large_toMove : large.toMove = .p1
  deficit_eq : large.deficit = small.deficit + j
  small_old : small.remaining ⊆ fourLiftOldCounters N
  k_not_small : k ∉ small.remaining
  remaining_eq : large.remaining = small.remaining ∪ newOpeningThreePacketCounters N k j

/-- Every legal small move is also present in the large mirror state. -/
theorem NewOpeningMirrorPhase.large_legal
    {N k x : ℕ} {small large : GamePosition}
    (hphase : NewOpeningMirrorPhase N k small large) (hx : small.LegalMove x) :
    large.LegalMove x := by
  rw [GamePosition.LegalMove, hphase.remaining_eq]
  exact Finset.mem_union_left _ hx

/-- A small old counter never belongs to the unmatched packet. -/
theorem NewOpeningMirrorPhase.not_mem_unmatched_packet
    {N k x : ℕ} {small large : GamePosition}
    (hphase : NewOpeningMirrorPhase N k small large) (hx : small.LegalMove x) :
    x ∉ newOpeningUnmatchedPacket N k := by
  have hxBounds := Finset.mem_Icc.mp (hphase.small_old hx)
  have hxNeK : x ≠ k := by
    intro h
    exact hphase.k_not_small (h ▸ hx)
  simp [newOpeningUnmatchedPacket, hxNeK, fourLiftTopCounters]
  omega

/-- A mirror-phase large position can still catch up. -/
theorem NewOpeningMirrorPhase.large_not_terminal
    {N k j : ℕ} {small large : GamePosition}
    (hphase : NewOpeningMirrorPhase N k small large)
    (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j) :
    ¬large.IsTerminal := by
  rcases hk with ⟨hkLower, hkUpper, hkN⟩
  rcases hj with ⟨hjLower, hjUpper, hjk⟩
  have hkPacket : k ∈ newOpeningUnmatchedPacket N k := by
    simp [newOpeningUnmatchedPacket]
  have hkLarge : large.LegalMove k := by
    rw [GamePosition.LegalMove, hphase.remaining_eq]
    exact Finset.mem_union_right _ hkPacket
  have hjTop : N + j ∈ fourLiftTopCounters N := by
    simp [fourLiftTopCounters]
    omega
  have hjNeTop : N + j ≠ N + k := by omega
  have hjPacket : N + j ∈ newOpeningUnmatchedPacket N k := by
    rw [newOpeningUnmatchedPacket, Finset.mem_insert]
    right
    exact Finset.mem_sdiff.mpr ⟨hjTop, by simpa using hjNeTop⟩
  have hjLarge : large.LegalMove (N + j) := by
    rw [GamePosition.LegalMove, hphase.remaining_eq]
    exact Finset.mem_union_right _ hjPacket
  have hkj : k ≠ N + j := by omega
  have hsubset : {k, N + j} ⊆ large.remaining := by
    intro counter hcounter
    simp only [Finset.mem_insert, Finset.mem_singleton] at hcounter
    rcases hcounter with rfl | rfl
    · exact hkLarge
    · exact hjLarge
  have hsum : ({k, N + j} : Finset ℕ).sum (fun x => x) ≤
      large.remaining.sum (fun x => x) := by
    apply Finset.sum_le_sum_of_subset_of_nonneg hsubset
    intros
    omega
  apply (GamePosition.not_is_terminal_iff large).2
  constructor
  · exact Finset.ne_empty_of_mem hjLarge
  · simp [hkj] at hsum
    rw [hphase.deficit_eq]
    have hdle := hphase.small_deficit_le
    omega

/-- Mirroring a pick that remains behind preserves the mirror phase. -/
theorem NewOpeningMirrorPhase.after_paired_pick_of_lt
    {N k x : ℕ} {small large : GamePosition}
    (hphase : NewOpeningMirrorPhase N k small large)
    (hx : small.LegalMove x) (hlt : x < small.deficit) :
    NewOpeningMirrorPhase N k (small.afterPick x) (large.afterPick x) := by
  have hxLarge := hphase.large_legal hx
  have hxPacket := hphase.not_mem_unmatched_packet hx
  have hxLargeLt : x < large.deficit := by rw [hphase.deficit_eq]; omega
  constructor
  · simpa [hphase.small_toMove] using GamePosition.to_move_after_pick_of_lt small x hlt
  · simpa [hphase.large_toMove] using GamePosition.to_move_after_pick_of_lt large x hxLargeLt
  · simp only [GamePosition.afterPick, Position.afterPick, hxLargeLt, hlt, if_true]
    rw [hphase.deficit_eq]
    omega
  · simp only [GamePosition.afterPick, Position.afterPick, hlt, if_true]
    omega
  · simp only [GamePosition.afterPick, Position.afterPick, hlt, if_true]
    exact (Nat.sub_le small.deficit x).trans hphase.small_deficit_le
  · intro counter hcounter
    apply hphase.small_old
    exact Finset.mem_of_mem_erase hcounter
  · simp only [GamePosition.remaining_after_pick, Finset.mem_erase]
    exact fun h => hphase.k_not_small h.2
  · rw [GamePosition.remaining_after_pick, GamePosition.remaining_after_pick,
      hphase.remaining_eq]
    ext counter
    simp only [Finset.mem_erase, Finset.mem_union]
    constructor
    · rintro ⟨hcounterNe, hsmall | hpacket⟩
      · exact Or.inl ⟨hcounterNe, hsmall⟩
      · exact Or.inr hpacket
    · rintro (⟨hcounterNe, hsmall⟩ | hpacket)
      · exact ⟨hcounterNe, Or.inl hsmall⟩
      · refine ⟨?_, Or.inr hpacket⟩
        intro hcounter
        subst counter
        exact hxPacket hpacket

/-- Mirroring the small catch-up pick enters the ready phase. -/
theorem NewOpeningMirrorPhase.after_paired_pick_of_le
    {N k x : ℕ} {small large : GamePosition}
    (hphase : NewOpeningMirrorPhase N k small large)
    (hx : small.LegalMove x) (hle : small.deficit ≤ x) :
    NewOpeningReadyPhase N k (small.afterPick x) (large.afterPick x) := by
  have hxBounds := Finset.mem_Icc.mp (hphase.small_old hx)
  have hxPacket := hphase.not_mem_unmatched_packet hx
  have hxLargeLt : x < large.deficit := by
    rw [hphase.deficit_eq]
    have := hphase.small_deficit_pos
    omega
  constructor
  · simpa [hphase.small_toMove] using GamePosition.to_move_after_pick_of_le small x hle
  · simpa [hphase.large_toMove] using GamePosition.to_move_after_pick_of_lt large x hxLargeLt
  · simp only [GamePosition.afterPick, Position.afterPick, hxLargeLt, if_true,
      show ¬x < small.deficit by omega, if_false]
    rw [hphase.deficit_eq]
    omega
  · simp only [GamePosition.afterPick, Position.afterPick,
      show ¬x < small.deficit by omega, if_false]
    have := hphase.small_deficit_pos
    omega
  · intro counter hcounter
    apply hphase.small_old
    exact Finset.mem_of_mem_erase hcounter
  · simp only [GamePosition.remaining_after_pick, Finset.mem_erase]
    exact fun h => hphase.k_not_small h.2
  · rw [GamePosition.remaining_after_pick, GamePosition.remaining_after_pick,
      hphase.remaining_eq]
    ext counter
    simp only [Finset.mem_erase, Finset.mem_union]
    constructor
    · rintro ⟨hcounterNe, hsmall | hpacket⟩
      · exact Or.inl ⟨hcounterNe, hsmall⟩
      · exact Or.inr hpacket
    · rintro (⟨hcounterNe, hsmall⟩ | hpacket)
      · exact ⟨hcounterNe, Or.inl hsmall⟩
      · refine ⟨?_, Or.inr hpacket⟩
        intro hcounter
        subst counter
        exact hxPacket hpacket

/-- The paired small/new opening roots begin in the mirror phase. -/
theorem new_opening_root_mirror_phase {N k : ℕ} (hk : ValidNewOpening N k) :
    NewOpeningMirrorPhase N k
      (GamePosition.afterOpening (Finset.Icc 1 N) k)
      (GamePosition.afterOpening (Finset.Icc 1 (N + 4)) (N + k)) := by
  rcases hk with ⟨hkLower, hkUpper, hkN⟩
  constructor
  · rfl
  · rfl
  · rfl
  · exact hkLower
  · exact le_rfl
  · intro counter hcounter
    change counter ∈ (Finset.Icc 1 N).erase k at hcounter
    exact Finset.mem_of_mem_erase hcounter
  · simp [GamePosition.afterOpening, CatchUp.afterOpening]
  · ext counter
    simp [GamePosition.afterOpening, CatchUp.afterOpening, newOpeningUnmatchedPacket,
      fourLiftTopCounters]
    omega

/-- Taking the distinct top reply turns every ready state into a three-packet state. -/
theorem NewOpeningReadyPhase.after_reply
    {N k j : ℕ} {small large : GamePosition}
    (hphase : NewOpeningReadyPhase N k small large)
    (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j) :
    ¬large.IsTerminal ∧ large.LegalMove (N + j) ∧
      NewOpeningThreePacket N k j small (large.afterPick (N + j)) := by
  rcases hk with ⟨hkLower, hkUpper, hkN⟩
  rcases hj with ⟨hjLower, hjUpper, hjk⟩
  have hjTop : N + j ∈ fourLiftTopCounters N := by
    simp [fourLiftTopCounters]
    omega
  have hjNe : N + j ≠ N + k := by omega
  have hjPacket : N + j ∈ newOpeningUnmatchedPacket N k := by
    rw [newOpeningUnmatchedPacket, Finset.mem_insert]
    right
    exact Finset.mem_sdiff.mpr ⟨hjTop, by simpa using hjNe⟩
  have hjLarge : large.LegalMove (N + j) := by
    rw [GamePosition.LegalMove, hphase.remaining_eq]
    exact Finset.mem_union_right _ hjPacket
  have hsum : N + j ≤ large.remaining.sum (fun x => x) := by
    exact Finset.single_le_sum (fun x _ => Nat.zero_le x) hjLarge
  have hnonterminal : ¬large.IsTerminal := by
    rw [GamePosition.IsTerminal]
    push_neg
    constructor
    · exact ⟨N + j, hjLarge⟩
    · rw [hphase.deficit_eq]
      omega
  refine ⟨hnonterminal, hjLarge, ?_⟩
  constructor
  · exact hphase.small_toMove
  · rw [GamePosition.to_move_after_pick_of_le]
    · simp [hphase.large_toMove]
    · rw [hphase.deficit_eq]
      omega
  · change (if N + j < large.deficit then large.deficit - (N + j)
        else N + j - large.deficit) = small.deficit + j
    rw [hphase.deficit_eq]
    simp only [show ¬N + j < N - small.deficit by omega, if_false]
    have hNsplit : N - small.deficit + small.deficit = N :=
      Nat.sub_add_cancel (Nat.le_of_lt hphase.small_deficit_lt)
    have hdecomp : N + j = (N - small.deficit) + (small.deficit + j) := by
      omega
    rw [hdecomp, Nat.add_sub_cancel_left]
  · exact hphase.small_old
  · exact hphase.k_not_small
  · rw [GamePosition.remaining_after_pick, hphase.remaining_eq]
    ext counter
    by_cases hsmall : counter ∈ small.remaining
    · have hcounterBounds := Finset.mem_Icc.mp (hphase.small_old hsmall)
      have hcounterNe : counter ≠ N + j := by omega
      simp [hsmall, hcounterNe]
    · by_cases hcounterK : counter = k
      · subst counter
        have hkNe : k ≠ N + j := by omega
        simp [hsmall, hkNe, newOpeningUnmatchedPacket, newOpeningThreePacketCounters]
      · simp [hsmall, hcounterK, newOpeningUnmatchedPacket,
          newOpeningThreePacketCounters]
        tauto

/--
Mirroring a safe player-two turn reduces new-opening safety to the three-packet endpoints.
-/
theorem NewOpeningMirrorPhase.can_avoid_loss
    {N k j : ℕ} {small large : GamePosition}
    (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j)
    (hphase : NewOpeningMirrorPhase N k small large)
    (hsmall : CanAvoidLoss .p2 small)
    (hend : ∀ {small' large' : GamePosition},
      NewOpeningThreePacket N k j small' large' →
        CanAvoidLoss .p2 small' → CanAvoidLoss .p2 large') :
    CanAvoidLoss .p2 large := by
  induction small using
      (measure fun position : GamePosition => position.remaining.card).wf.induction
      generalizing large with
  | h small ih =>
      have hsmallTerminal : ¬small.IsTerminal := by
        intro hterminal
        have hsafeTerminal :=
          (can_avoid_loss_iff_terminal_outcome_ne_loss .p2 small hterminal).1 hsmall
        have houtcome : small.terminalOutcomeFor .p2 = .loss := by
          rw [← hphase.small_toMove]
          exact small.terminal_outcome_for_to_move_eq_loss
            (Nat.ne_of_gt hphase.small_deficit_pos)
        exact hsafeTerminal houtcome
      obtain ⟨x, hx, hxSafe⟩ :=
        (can_avoid_loss_iff_exists_safe_move .p2 small hsmallTerminal
          hphase.small_toMove).1 hsmall
      have hlargeTerminal := hphase.large_not_terminal hk hj
      apply (can_avoid_loss_iff_exists_safe_move .p2 large hlargeTerminal
        hphase.large_toMove).2
      refine ⟨x, hphase.large_legal hx, ?_⟩
      by_cases hlt : x < small.deficit
      · have hphase' := hphase.after_paired_pick_of_lt hx hlt
        exact ih (small.afterPick x) (small.remaining_card_after_pick_lt x hx)
          hphase' hxSafe
      · have hle : small.deficit ≤ x := by omega
        have hready := hphase.after_paired_pick_of_le hx hle
        obtain ⟨hlargeAfterTerminal, hjLegal, hpacket⟩ := hready.after_reply hk hj
        apply (can_avoid_loss_iff_exists_safe_move .p2 (large.afterPick x)
          hlargeAfterTerminal hready.large_toMove).2
        exact ⟨N + j, hjLegal, hend hpacket hxSafe⟩

/-- A three-packet endpoint has exactly the opening counter pending between its old supports. -/
theorem NewOpeningThreePacket.pending_counters
    {N k j : ℕ} {small large : GamePosition}
    (hphase : NewOpeningThreePacket N k j small large)
    (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j) :
    fourLiftPendingCounters N small large = {k} := by
  rcases hk with ⟨hkLower, hkUpper, hkN⟩
  rcases hj with ⟨hjLower, hjUpper, hjk⟩
  ext counter
  by_cases hsmall : counter ∈ small.remaining
  · have hbounds := Finset.mem_Icc.mp (hphase.small_old hsmall)
    have hne : counter ≠ k := by
      intro h
      exact hphase.k_not_small (h ▸ hsmall)
    simp [fourLiftPendingCounters, GamePosition.fourLiftOldRemaining,
      fourLiftOldCounters, hphase.remaining_eq, newOpeningThreePacketCounters,
      fourLiftTopCounters, hsmall, hne, Finset.mem_symmDiff]
  · by_cases hcounter : counter = k
    · subst counter
      simp [fourLiftPendingCounters, GamePosition.fourLiftOldRemaining,
        fourLiftOldCounters, hphase.remaining_eq, newOpeningThreePacketCounters,
        fourLiftTopCounters, hsmall, Finset.mem_symmDiff]
      omega
    · simp [fourLiftPendingCounters, GamePosition.fourLiftOldRemaining,
        fourLiftOldCounters, hphase.remaining_eq, newOpeningThreePacketCounters,
        fourLiftTopCounters, hsmall, hcounter, Finset.mem_symmDiff]
      omega

/-- A three-packet endpoint has consumed exactly the two selected added counters. -/
theorem NewOpeningThreePacket.consumed_top_counters
    {N k j : ℕ} {small large : GamePosition}
    (hphase : NewOpeningThreePacket N k j small large)
    (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j) :
    fourLiftConsumedTopCounters N large = {N + k, N + j} := by
  rcases hk with ⟨hkLower, hkUpper, hkN⟩
  rcases hj with ⟨hjLower, hjUpper, hjk⟩
  ext counter
  by_cases hsmall : counter ∈ small.remaining
  · have hbounds := Finset.mem_Icc.mp (hphase.small_old hsmall)
    simp [fourLiftConsumedTopCounters, hphase.remaining_eq,
      newOpeningThreePacketCounters, fourLiftTopCounters, hsmall]
    omega
  · simp [fourLiftConsumedTopCounters, hphase.remaining_eq,
      newOpeningThreePacketCounters, fourLiftTopCounters, hsmall]
    omega

/-- The new-opening ledger exactly represents every three-packet endpoint. -/
theorem NewOpeningThreePacket.consistent_new_opening_ledger
    {N k j : ℕ} {small large : GamePosition}
    (hphase : NewOpeningThreePacket N k j small large)
    (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j) :
    (FourLiftLedger.newOpening N k).Consistent N small large := by
  have hpending := hphase.pending_counters hk hj
  have htop := hphase.consumed_top_counters hk hj
  rcases hk with ⟨hkLower, hkUpper, hkN⟩
  rcases hj with ⟨hjLower, hjUpper, hjk⟩
  have htopNe : N + k ≠ N + j := by omega
  have htopSum : signedCounterSum {N + k, N + j} {N + k} = (k : ℤ) - j := by
    unfold signedCounterSum
    rw [Finset.sum_insert]
    · simp
      omega
    · simpa using htopNe
  constructor
  · refine ⟨hphase.small_old, ?_, ?_, ?_, ?_, ?_⟩
    · intro counter hcounter
      rw [hphase.remaining_eq] at hcounter
      rcases Finset.mem_union.mp hcounter with hsmall | hpacket
      · have hbounds := Finset.mem_Icc.mp (hphase.small_old hsmall)
        simp
        omega
      · simp [newOpeningThreePacketCounters, fourLiftTopCounters] at hpacket
        simp
        omega
    · rw [htop]
      simp [FourLiftLedger.newOpening]
    · simp [FourLiftLedger.newOpening]
    · rw [hpending]
      simp
    · simp [FourLiftLedger.newOpening]
  · rw [FourLiftLedger.value, htop, hpending]
    simp only [FourLiftLedger.newOpening]
    rw [htopSum]
    simp [signedCounterSum, GamePosition.p1ScoreBalance,
      hphase.small_toMove, hphase.large_toMove]
    rw [hphase.deficit_eq]
    omega

/-- A uniform distinct added-counter reply for each of the four new openings. -/
def canonicalNewOpeningReply (k : ℕ) : ℕ :=
  if k = 1 then 2 else 1

/-- The canonical reply is valid for every opening offset. -/
theorem valid_canonical_new_opening_reply (k : ℕ) :
    ValidNewOpeningReply k (canonicalNewOpeningReply k) := by
  by_cases hk : k = 1
  · subst k
    simp [ValidNewOpeningReply, canonicalNewOpeningReply]
  · have hk' : 1 ≠ k := Ne.symm hk
    simp [ValidNewOpeningReply, canonicalNewOpeningReply, hk, hk']

/-- Safety transfer for every endpoint in one three-packet family. -/
def NewOpeningThreePacketTransfer (N k j : ℕ) : Prop :=
  ∀ {small large : GamePosition},
    NewOpeningThreePacket N k j small large →
      CanAvoidLoss .p2 small → CanAvoidLoss .p2 large

/-- The three-packet endpoint transfer implies safety at its paired new-opening root. -/
theorem can_avoid_loss_after_new_opening
    {N k j : ℕ} (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j)
    (htransfer : NewOpeningThreePacketTransfer N k j)
    (hsmall : CanAvoidLoss .p2 (GamePosition.afterOpening (Finset.Icc 1 N) k)) :
    CanAvoidLoss .p2
      (GamePosition.afterOpening (Finset.Icc 1 (N + 4)) (N + k)) := by
  exact NewOpeningMirrorPhase.can_avoid_loss hk hj
    (new_opening_root_mirror_phase hk) hsmall htransfer

/-- Canonical three-packet endpoint transfers for all four new-opening offsets. -/
def CanonicalNewOpeningThreePacketTransfers (N : ℕ) : Prop :=
  ∀ k ∈ Finset.Icc 1 4,
    NewOpeningThreePacketTransfer N k (canonicalNewOpeningReply k)

/-- Canonical endpoint transfers imply the player-two new-opening transfer API. -/
theorem CanonicalNewOpeningThreePacketTransfers.to_new_opening_transfer
    {N : ℕ} (htransfers : CanonicalNewOpeningThreePacketTransfers N) (hN : 4 ≤ N) :
    FourLiftNewOpeningTransfer N .p2 := by
  intro k hk hsmall
  have hkBounds := Finset.mem_Icc.mp hk
  have hkValid : ValidNewOpening N k :=
    ⟨hkBounds.1, hkBounds.2, hkBounds.2.trans hN⟩
  exact can_avoid_loss_after_new_opening hkValid
    (valid_canonical_new_opening_reply k) (htransfers k hk) hsmall

/-- Every three-packet endpoint is related by the persistent new-opening ledger. -/
theorem NewOpeningThreePacket.related
    {N k j : ℕ} {small large : GamePosition}
    (hphase : NewOpeningThreePacket N k j small large)
    (hk : ValidNewOpening N k) (hj : ValidNewOpeningReply k j) :
    FourLiftGhost.Related N ⟨small, FourLiftLedger.newOpening N k⟩ large := by
  exact hphase.consistent_new_opening_ledger hk hj
