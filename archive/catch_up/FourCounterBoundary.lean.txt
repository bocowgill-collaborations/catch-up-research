import catch_up.FourCounterAttractor

/-!
# Catch-Up four-counter forcing-basin boundaries

This research module inspects opponent choices on the boundary of a safe top-only forcing basin
while retaining only play that preserves non-loss for the chosen owner. A mixed opponent boundary
node lies outside the basin but has both a move into the basin and a move remaining outside it. The
latter moves are the local obstructions to forcing the target endgame. The analysis also tests the
smallest natural enlargement suggested by this boundary: every safe state with at most one old
counter.
-/

namespace CatchUpSearch.FourCounterBoundary

open CatchUp
open CatchUpSearch.ExactSolver
open CatchUpSearch.FourCounterAttractor

/-- Moves from one node whose children lie inside the forcing basin. -/
def enteringMoves (cache : BasinCache) (node : SolvedNode) : List MoveEvaluation :=
  node.moves.filter fun move => basinContains cache move.child

/-- Moves from one node whose children remain outside the forcing basin. -/
def escapingMoves (cache : BasinCache) (node : SolvedNode) : List MoveEvaluation :=
  node.moves.filter fun move => !basinContains cache move.child

/-- An opponent node just outside the basin with both entering and escaping choices. -/
def isMixedOpponentBoundary (owner : Player) (cache : BasinCache) (node : SolvedNode) : Bool :=
  !basinContains cache node.state &&
    decide (node.state.toMove ≠ owner) &&
    !(enteringMoves cache node).isEmpty &&
    !(escapingMoves cache node).isEmpty

/-- Whether one move preserves non-loss for the fixed owner. -/
def moveSafeFor (owner : Player) (node : SolvedNode) (move : MoveEvaluation) : Bool :=
  StrategyExtraction.outcomeSafeFor owner node.state move.outcome

/-- Safe outside-basin nodes reached while the opponent consistently chooses an escape. -/
abbrev AvoidanceIndex := Std.HashMap CacheKey SolvedNode

/-- Avoidance traversal with memoization and explicit failure. -/
abbrev AvoidanceM := StateT AvoidanceIndex (Except String)

/-- Traverse the outside-basin graph generated by safe owner moves and safe opponent escapes. -/
def collectAvoidanceAux (owner : Player) (nodes : SolvedIndex) (cache : BasinCache) :
    ℕ → State → AvoidanceM Unit
  | 0, state => do
      if (← get)[state.cacheKey]?.isSome then
        return
      throw s!"avoidance traversal fuel exhausted at {repr state}"
  | fuel + 1, state => do
      if (← get)[state.cacheKey]?.isSome then
        return
      if basinContains cache state then
        throw s!"avoidance traversal entered the forcing basin at {repr state}"
      let node ← match nodes[state.cacheKey]? with
        | none => throw s!"avoidance traversal is missing state {repr state}"
        | some node => pure node
      if !StrategyExtraction.outcomeSafeFor owner node.state node.outcome then
        throw s!"avoidance traversal reached a state unsafe for the owner at {repr state}"
      modify fun visited => visited.insert state.cacheKey node
      if node.moves.isEmpty then
        return
      let safeMoves := node.moves.filter (moveSafeFor owner node)
      if safeMoves.isEmpty then
        throw s!"safe nonterminal node has no safe move at {repr state}"
      if state.toMove = owner then
        for move in safeMoves do
          if basinContains cache move.child then
            throw s!"outside owner node has a child in the forcing basin at {repr state}"
          collectAvoidanceAux owner nodes cache fuel move.child
      else
        let escapes := safeMoves.filter fun move => !basinContains cache move.child
        if escapes.isEmpty then
          throw s!"outside opponent node has no escape from the forcing basin at {repr state}"
        for move in escapes do
          collectAvoidanceAux owner nodes cache fuel move.child

/-- Collect safe outside nodes reachable under basin avoidance. -/
def collectAvoidance (owner : Player) (output : SolverOutput) (cache : BasinCache)
    (root : State) : Except String (List SolvedNode) := do
  let nodes := indexSolvedNodes output
  let (_, visited) ←
    (collectAvoidanceAux owner nodes cache (output.gameSize + 1) root).run {}
  pure <| (visited.toList.map fun entry => entry.2).mergeSort fun left right =>
    stateLE left.state right.state

/-- A safe mixed opponent boundary relevant to a non-loss strategy. -/
def isSafeMixedOpponentBoundary (owner : Player) (cache : BasinCache)
    (node : SolvedNode) : Bool :=
  StrategyExtraction.outcomeSafeFor owner node.state node.outcome &&
    isMixedOpponentBoundary owner cache node

/-- One mixed boundary state and the opponent choices on its two sides. -/
structure BoundaryWitness where
  state : State
  depth : ℕ
  oldCounters : ℕ
  topMask : ℕ
  enters : List MoveEvaluation
  escapes : List MoveEvaluation
deriving Repr

/-- Number of counters removed after the opening move on a path to `state`. -/
def depthAfterOpeningOne (N : ℕ) (state : State) : ℕ :=
  (N + 3) - (availableMoves (N + 4) state.remainingMask).length

/-- Record the local data needed to inspect one mixed boundary state. -/
def makeWitness (N : ℕ) (cache : BasinCache) (node : SolvedNode) : BoundaryWitness :=
  { state := node.state
    depth := depthAfterOpeningOne N node.state
    oldCounters := oldCounterCount N node.state
    topMask := PairedTrace.topMask N node.state
    enters := enteringMoves cache node
    escapes := escapingMoves cache node }

/-- Order boundary nodes from fewest remaining old counters, then by normalized state. -/
def boundaryLE (N : ℕ) (left right : SolvedNode) : Bool :=
  let leftOld := oldCounterCount N left.state
  let rightOld := oldCounterCount N right.state
  if leftOld < rightOld then true
  else if rightOld < leftOld then false
  else stateLE left.state right.state

/-- Order boundary nodes by distance from the opening-`1` root, then by normalized state. -/
def boundaryDepthLE (N : ℕ) (left right : SolvedNode) : Bool :=
  let leftDepth := depthAfterOpeningOne N left.state
  let rightDepth := depthAfterOpeningOne N right.state
  if leftDepth < rightDepth then true
  else if rightDepth < leftDepth then false
  else stateLE left.state right.state

/-- Nonzero move counts grouped by the selected counter. -/
def counterHistogram (gameSize : ℕ) (moves : List MoveEvaluation) : List (ℕ × ℕ) :=
  (List.range gameSize).filterMap fun offset =>
    let counter := offset + 1
    let count := (moves.filter fun move => decide (move.counter = counter)).length
    if count = 0 then none else some (counter, count)

/-- Aggregate evidence about the basin boundary reachable from an opening-`1` root. -/
structure BoundarySummary where
  smallN : ℕ
  owner : Player
  graphNodes : ℕ
  basinNodes : ℕ
  oneOldTargetNodes : ℕ
  oneOldBasinNodes : ℕ
  rootInOneOldBasin : Bool
  explicitTargetNodes : ℕ
  explicitBasinNodes : ℕ
  rootInExplicitBasin : Bool
  explicitReachableSafeOutsideNodes : ℕ
  explicitReachableSafeBoundaryNodes : ℕ
  explicitRootBoundary : Option BoundaryWitness
  explicitClosestOneOldWitnesses : List BoundaryWitness
  reachableSafeOutsideNodes : ℕ
  globalSafeBoundaryNodes : ℕ
  reachableSafeBoundaryNodes : ℕ
  terminalSafeOutsideNodes : ℕ
  minimumBoundaryOldCounters : Option ℕ
  minimumPositiveBoundaryOldCounters : Option ℕ
  rootBoundary : Option BoundaryWitness
  enteringCounterHistogram : List (ℕ × ℕ)
  escapingCounterHistogram : List (ℕ × ℕ)
  earliestWitnesses : List BoundaryWitness
  closestPositiveOldWitnesses : List BoundaryWitness
  topOnlyWitnesses : List BoundaryWitness
deriving Repr

/-- Analyze the reachable boundary of one owner's failed safe-top-only forcing basin. -/
def analyzeOutput (N : ℕ) (owner : Player) (output : SolverOutput)
    (witnessLimit : ℕ := 4) : Except String BoundarySummary := do
  if !topOnlyClassificationAgrees N owner output then
    throw s!"proof-backed top-only classification disagrees with the solver at N={N} for {repr owner}"
  checkTargetCertificates .safeTopOnly N owner output
  let cache ← computeBasin .safeTopOnly N owner output
  if !basinValid .safeTopOnly N owner output cache then
    throw s!"safe-top-only basin validation failed at N={N} for {repr owner}"
  let root ← match output.openingRoots.find? fun candidate => decide (candidate.1 = 1) with
    | none => throw s!"opening 1 is absent from the solved output at N={N}"
    | some root => pure root.2
  if basinContains cache root then
    throw s!"opening-1 root already lies in the safe-top-only basin at N={N} for {repr owner}"
  checkTargetCertificates (.safeAtMostOld 1) N owner output
  let oneOldCache ← computeBasin (.safeAtMostOld 1) N owner output
  if !basinValid (.safeAtMostOld 1) N owner output oneOldCache then
    throw s!"safe one-old basin validation failed at N={N} for {repr owner}"
  checkTargetCertificates .safeTopOnlyOrOneOldPair N owner output
  let explicitCache ← computeBasin .safeTopOnlyOrOneOldPair N owner output
  if !basinValid .safeTopOnlyOrOneOldPair N owner output explicitCache then
    throw s!"explicit one-old-pair basin validation failed at N={N} for {repr owner}"
  let reachable ← collectAvoidance owner output cache root
  let explicitReachable ←
    if basinContains explicitCache root then pure [] else collectAvoidance owner output explicitCache root
  let globalBoundary := output.nodes.filter (isSafeMixedOpponentBoundary owner cache)
  let reachableBoundary := reachable.filter (isSafeMixedOpponentBoundary owner cache)
  let explicitReachableBoundary :=
    explicitReachable.filter (isSafeMixedOpponentBoundary owner explicitCache)
  let orderedBoundary := reachableBoundary.mergeSort (boundaryLE N)
  let orderedPositiveBoundary :=
    (reachableBoundary.filter fun node => decide (0 < oldCounterCount N node.state)).mergeSort
      (boundaryLE N)
  let orderedTopOnlyBoundary :=
    (reachableBoundary.filter fun node => decide (oldCounterCount N node.state = 0)).mergeSort
      fun left right => stateLE left.state right.state
  let orderedByDepth := reachableBoundary.mergeSort (boundaryDepthLE N)
  let terminalOutside := reachable.filter fun node => node.moves.isEmpty
  let rootBoundary :=
    match findNode? output root with
    | some node =>
        if isSafeMixedOpponentBoundary owner cache node then some (makeWitness N cache node) else none
    | none => none
  let explicitRootBoundary :=
    match findNode? output root with
    | some node =>
        if isSafeMixedOpponentBoundary owner explicitCache node then
          some (makeWitness N explicitCache node)
        else none
    | none => none
  let explicitOrderedOneOld :=
    (explicitReachableBoundary.filter fun node => decide (oldCounterCount N node.state = 1)).mergeSort
      fun left right => stateLE left.state right.state
  let allEntering := reachableBoundary.flatMap (enteringMoves cache)
  let allEscaping := reachableBoundary.flatMap (escapingMoves cache)
  pure
    { smallN := N
      owner
      graphNodes := output.nodes.length
      basinNodes := (output.nodes.filter fun node => basinContains cache node.state).length
      oneOldTargetNodes :=
        (output.nodes.filter (isTarget (.safeAtMostOld 1) N owner)).length
      oneOldBasinNodes :=
        (output.nodes.filter fun node => basinContains oneOldCache node.state).length
      rootInOneOldBasin := basinContains oneOldCache root
      explicitTargetNodes :=
        (output.nodes.filter (isTarget .safeTopOnlyOrOneOldPair N owner)).length
      explicitBasinNodes :=
        (output.nodes.filter fun node => basinContains explicitCache node.state).length
      rootInExplicitBasin := basinContains explicitCache root
      explicitReachableSafeOutsideNodes := explicitReachable.length
      explicitReachableSafeBoundaryNodes := explicitReachableBoundary.length
      explicitRootBoundary
      explicitClosestOneOldWitnesses :=
        (explicitOrderedOneOld.take witnessLimit).map (makeWitness N explicitCache)
      reachableSafeOutsideNodes := reachable.length
      globalSafeBoundaryNodes := globalBoundary.length
      reachableSafeBoundaryNodes := reachableBoundary.length
      terminalSafeOutsideNodes := terminalOutside.length
      minimumBoundaryOldCounters := orderedBoundary.head?.map fun node => oldCounterCount N node.state
      minimumPositiveBoundaryOldCounters :=
        orderedPositiveBoundary.head?.map fun node => oldCounterCount N node.state
      rootBoundary
      enteringCounterHistogram := counterHistogram output.gameSize allEntering
      escapingCounterHistogram := counterHistogram output.gameSize allEscaping
      earliestWitnesses := (orderedByDepth.take witnessLimit).map (makeWitness N cache)
      closestPositiveOldWitnesses :=
        (orderedPositiveBoundary.take witnessLimit).map (makeWitness N cache)
      topOnlyWitnesses := (orderedTopOnlyBoundary.take witnessLimit).map (makeWitness N cache) }

/-- Solve an opening-`1` graph and inspect one owner's failed safe-top-only basin boundary. -/
def analyze (N : ℕ) (owner : Player) (witnessLimit : ℕ := 4) :
    Except String BoundarySummary := do
  let output ← PairedTrace.solveOpeningGraph (N + 4) 1
  analyzeOutput N owner output witnessLimit

end CatchUpSearch.FourCounterBoundary
