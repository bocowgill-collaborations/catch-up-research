import catch_up.DenseHoleReduction

/-!
# Shifted-tail reduction for the dense minimum-move candidate

This scratch module gives a computation-free structural reduction of minimum-move safety to one
continuation-side and one catch-side family of shifted-tail positions.
-/

namespace CatchUp

/-- Continuation-side shifted-tail assertion for the minimum-move reduction.  The term
`m - 1` counts the forced missing prefix `1, ..., m - 1`; the other summand counts the
additional holes above the deleted minimum. -/
def ShiftedTailContinuationNonloss (N : ℕ) : Prop :=
  ∀ (m : ℕ) (remaining : Finset ℕ) (deficit : ℕ),
    1 ≤ m →
    m ≤ N →
    remaining ⊆ Finset.Icc (m + 1) N →
    ((Finset.Icc (m + 1) N) \ remaining).card + (m - 1) ≤ 3 →
    1 ≤ deficit →
    deficit ≤ N - m →
    ¬ remaining.sum (fun x => x) < deficit →
    Even (deficit + remaining.sum (fun x => x)) →
    (⟨remaining, deficit⟩ : Position).value ≠ .loss

/-- Catch-side shifted-tail assertion for the minimum-move reduction. -/
def ShiftedTailCatchNonwin (N : ℕ) : Prop :=
  ∀ (m : ℕ) (remaining : Finset ℕ) (deficit : ℕ),
    1 ≤ m →
    m ≤ N →
    remaining ⊆ Finset.Icc (m + 1) N →
    ((Finset.Icc (m + 1) N) \ remaining).card + (m - 1) ≤ 3 →
    deficit ≤ m →
    Even (deficit + remaining.sum (fun x => x)) →
    (⟨remaining, deficit⟩ : Position).value ≠ .win

private theorem erase_minimum_is_shifted_tail
    (N : ℕ) (remaining : Finset ℕ) (m : ℕ)
    (hsub : remaining ⊆ Finset.Icc 1 N)
    (hholes : ((Finset.Icc 1 N) \ remaining).card ≤ 3)
    (hm : m ∈ remaining)
    (hleast : ∀ x ∈ remaining, m ≤ x) :
    1 ≤ m ∧
      m ≤ N ∧
        remaining.erase m ⊆ Finset.Icc (m + 1) N ∧
          ((Finset.Icc (m + 1) N) \ (remaining.erase m)).card + (m - 1) ≤ 3 := by
  have hmBounds := Finset.mem_Icc.mp (hsub hm)
  refine ⟨hmBounds.1, hmBounds.2, ?_, ?_⟩
  · intro x hx
    have hxRemaining := Finset.mem_of_mem_erase hx
    have hxBounds := Finset.mem_Icc.mp (hsub hxRemaining)
    have hxm : x ≠ m := (Finset.mem_erase.mp hx).1
    have hmx : m < x := lt_of_le_of_ne (hleast x hxRemaining) (Ne.symm hxm)
    exact Finset.mem_Icc.mpr ⟨by omega, hxBounds.2⟩
  · let lower := Finset.Icc 1 (m - 1)
    let upper := (Finset.Icc (m + 1) N) \ (remaining.erase m)
    have hlowerUpper : Disjoint lower upper := by
      rw [Finset.disjoint_left]
      intro x hxLower hxUpper
      have hxLowerBounds := Finset.mem_Icc.mp hxLower
      have hxUpperBounds := Finset.mem_Icc.mp (Finset.mem_sdiff.mp hxUpper).1
      omega
    have hunion : lower ∪ upper ⊆ (Finset.Icc 1 N) \ remaining := by
      intro x hx
      rcases Finset.mem_union.mp hx with hxLower | hxUpper
      · have hxBounds := Finset.mem_Icc.mp hxLower
        apply Finset.mem_sdiff.mpr
        constructor
        · exact Finset.mem_Icc.mpr ⟨hxBounds.1, by omega⟩
        · intro hxRemaining
          have hmx := hleast x hxRemaining
          omega
      · have hxTail := (Finset.mem_sdiff.mp hxUpper).1
        have hxNotErase := (Finset.mem_sdiff.mp hxUpper).2
        have hxBounds := Finset.mem_Icc.mp hxTail
        apply Finset.mem_sdiff.mpr
        constructor
        · exact Finset.mem_Icc.mpr ⟨by omega, hxBounds.2⟩
        · intro hxRemaining
          apply hxNotErase
          exact Finset.mem_erase.mpr ⟨by omega, hxRemaining⟩
    have hcard := Finset.card_le_card hunion
    have hlowerCard : lower.card = m - 1 := by
      simp [lower, Nat.card_Icc]
    rw [Finset.card_union_of_disjoint hlowerUpper, hlowerCard] at hcard
    simpa only [upper, Nat.add_comm] using hcard.trans hholes

/-- The two shifted-tail assertions imply safety of the minimum counter in every qualifying
dense three-hole position.  No lower bound on `N` is needed for this purely structural direction. -/
@[category API, AMS 91]
theorem dense_three_hole_minimum_move_safe_of_shifted_tail
    (N : ℕ)
    (hcontinue : ShiftedTailContinuationNonloss N)
    (hcatch : ShiftedTailCatchNonwin N) :
    DenseThreeHoleMinimumMoveSafe N := by
  intro position hsub hholes hdN hcanCatch heven m hm hleast
  obtain ⟨hmPos, hmN, hchildSub, hchildHoles⟩ :=
    erase_minimum_is_shifted_tail N position.remaining m hsub hholes hm hleast
  let child := position.afterPick m
  have hchildEven :
      Even (child.deficit + child.remaining.sum (fun x => x)) := by
    rw [Nat.even_iff] at heven ⊢
    have hmod := Position.mod_eq_two_deficit_add_sum_after_pick position hm
    change (child.deficit + child.remaining.sum (fun x => x)) % 2 =
      (position.deficit + position.remaining.sum (fun x => x)) % 2 at hmod
    exact hmod.trans heven
  have hsumErase :
      (position.remaining.erase m).sum (fun x => x) + m =
        position.remaining.sum (fun x => x) := by
    simpa using Finset.sum_erase_add position.remaining (fun x => x) hm
  by_cases hlt : m < position.deficit
  · have hchild : child = ⟨position.remaining.erase m, position.deficit - m⟩ := by
      simp [child, Position.afterPick, hlt]
    have hdeficit : position.deficit - m ≤ N - m := by omega
    have hdeficitPos : 1 ≤ position.deficit - m := by omega
    have hchildCatch :
        ¬ (position.remaining.erase m).sum (fun x => x) < position.deficit - m := by
      omega
    have hvalue :
        (⟨position.remaining.erase m, position.deficit - m⟩ : Position).value ≠ .loss :=
      hcontinue m (position.remaining.erase m) (position.deficit - m)
        hmPos hmN hchildSub hchildHoles hdeficitPos hdeficit hchildCatch
          (by simpa [hchild] using hchildEven)
    rw [Position.moveOutcome]
    simp only [hlt, if_true]
    change child.value ≠ .loss
    rw [hchild]
    exact hvalue
  · have hdeficit : m - position.deficit ≤ m := by omega
    have hvalue :
        (⟨position.remaining.erase m, m - position.deficit⟩ : Position).value ≠ .win :=
      hcatch m (position.remaining.erase m) (m - position.deficit)
        hmPos hmN hchildSub hchildHoles hdeficit (by
          simpa [child, Position.afterPick, hlt] using hchildEven)
    rw [Position.moveOutcome]
    simp only [hlt, if_false]
    rw [Outcome.neg_ne_loss_iff]
    simpa [child, Position.afterPick, hlt] using hvalue

private theorem insert_shifted_minimum_is_dense
    (N : ℕ) (m : ℕ) (remaining : Finset ℕ)
    (hmPos : 1 ≤ m) (hmN : m ≤ N)
    (hsub : remaining ⊆ Finset.Icc (m + 1) N)
    (hholes : ((Finset.Icc (m + 1) N) \ remaining).card + (m - 1) ≤ 3) :
    insert m remaining ⊆ Finset.Icc 1 N ∧
      ((Finset.Icc 1 N) \ insert m remaining).card ≤ 3 ∧
        m ∈ insert m remaining ∧
          (∀ x ∈ insert m remaining, m ≤ x) ∧
            m ∉ remaining := by
  have hmNotRemaining : m ∉ remaining := by
    intro hmRemaining
    have hmTail := Finset.mem_Icc.mp (hsub hmRemaining)
    omega
  have hparentSub : insert m remaining ⊆ Finset.Icc 1 N := by
    intro x hx
    rcases Finset.mem_insert.mp hx with rfl | hxRemaining
    · exact Finset.mem_Icc.mpr ⟨hmPos, hmN⟩
    · have hxBounds := Finset.mem_Icc.mp (hsub hxRemaining)
      exact Finset.mem_Icc.mpr ⟨by omega, hxBounds.2⟩
  have hleast : ∀ x ∈ insert m remaining, m ≤ x := by
    intro x hx
    rcases Finset.mem_insert.mp hx with rfl | hxRemaining
    · exact le_rfl
    · have hxBounds := Finset.mem_Icc.mp (hsub hxRemaining)
      omega
  let lower := Finset.Icc 1 (m - 1)
  let upper := (Finset.Icc (m + 1) N) \ remaining
  have hlowerUpper : Disjoint lower upper := by
    rw [Finset.disjoint_left]
    intro x hxLower hxUpper
    have hxLowerBounds := Finset.mem_Icc.mp hxLower
    have hxUpperBounds := Finset.mem_Icc.mp (Finset.mem_sdiff.mp hxUpper).1
    omega
  have hdecompose :
      (Finset.Icc 1 N) \ insert m remaining = lower ∪ upper := by
    ext x
    constructor
    · intro hx
      have hxBounds := Finset.mem_Icc.mp (Finset.mem_sdiff.mp hx).1
      have hxNotParent := (Finset.mem_sdiff.mp hx).2
      have hxm : x ≠ m := by
        intro hxm
        subst x
        exact hxNotParent (by simp)
      by_cases hxmLower : x < m
      · apply Finset.mem_union.mpr
        left
        exact Finset.mem_Icc.mpr ⟨hxBounds.1, by omega⟩
      · apply Finset.mem_union.mpr
        right
        apply Finset.mem_sdiff.mpr
        constructor
        · exact Finset.mem_Icc.mpr ⟨by omega, hxBounds.2⟩
        · intro hxRemaining
          exact hxNotParent (by simp [hxRemaining])
    · intro hx
      rcases Finset.mem_union.mp hx with hxLower | hxUpper
      · have hxBounds := Finset.mem_Icc.mp hxLower
        apply Finset.mem_sdiff.mpr
        constructor
        · exact Finset.mem_Icc.mpr ⟨hxBounds.1, by omega⟩
        · intro hxParent
          rcases Finset.mem_insert.mp hxParent with hxm | hxRemaining
          · omega
          · have hxTail := Finset.mem_Icc.mp (hsub hxRemaining)
            omega
      · have hxTail := (Finset.mem_sdiff.mp hxUpper).1
        have hxNotRemaining := (Finset.mem_sdiff.mp hxUpper).2
        have hxBounds := Finset.mem_Icc.mp hxTail
        apply Finset.mem_sdiff.mpr
        constructor
        · exact Finset.mem_Icc.mpr ⟨by omega, hxBounds.2⟩
        · intro hxParent
          rcases Finset.mem_insert.mp hxParent with hxm | hxRemaining
          · omega
          · exact hxNotRemaining hxRemaining
  have hlowerCard : lower.card = m - 1 := by
    simp [lower, Nat.card_Icc]
  have hparentHoles : ((Finset.Icc 1 N) \ insert m remaining).card ≤ 3 := by
    rw [hdecompose, Finset.card_union_of_disjoint hlowerUpper, hlowerCard]
    simpa only [upper, Nat.add_comm] using hholes
  exact ⟨hparentSub, hparentHoles, by simp, hleast, hmNotRemaining⟩

/-- Minimum-move safety also recovers both shifted-tail assertions.  Hence the structural
reduction is an exact equivalence, rather than merely a sufficient strengthening. -/
@[category API, AMS 91]
theorem shifted_tail_iff_dense_three_hole_minimum_move_safe (N : ℕ) :
    ShiftedTailContinuationNonloss N ∧ ShiftedTailCatchNonwin N ↔
      DenseThreeHoleMinimumMoveSafe N := by
  constructor
  · rintro ⟨hcontinue, hcatch⟩
    exact dense_three_hole_minimum_move_safe_of_shifted_tail N hcontinue hcatch
  · intro hminimum
    constructor
    · intro m remaining deficit hmPos hmN hsub hholes hdeficitPos hdeficitN
        hcanCatch heven
      obtain ⟨hparentSub, hparentHoles, hmParent, hleast, hmNotRemaining⟩ :=
        insert_shifted_minimum_is_dense N m remaining hmPos hmN hsub hholes
      let parent : Position := ⟨insert m remaining, deficit + m⟩
      have hsumParent :
          parent.remaining.sum (fun x => x) = remaining.sum (fun x => x) + m := by
        simp [parent, Finset.sum_insert hmNotRemaining, Nat.add_comm]
      have hparentDeficit : parent.deficit ≤ N := by
        change deficit + m ≤ N
        omega
      have hparentCatch :
          ¬ parent.remaining.sum (fun x => x) < parent.deficit := by
        rw [hsumParent]
        change ¬ remaining.sum (fun x => x) + m < deficit + m
        omega
      have hparentEven :
          Even (parent.deficit + parent.remaining.sum (fun x => x)) := by
        rcases heven with ⟨t, ht⟩
        refine ⟨t + m, ?_⟩
        simp only [parent, hsumParent]
        change deficit + m + (remaining.sum (fun x => x) + m) =
          (t + m) + (t + m)
        omega
      have hsafe := hminimum parent (by simpa [parent] using hparentSub)
        (by simpa [parent] using hparentHoles) hparentDeficit hparentCatch hparentEven
        m (by simp [parent]) (by simpa [parent] using hleast)
      have hlt : m < parent.deficit := by
        change m < deficit + m
        omega
      rw [Position.moveOutcome] at hsafe
      simp only [hlt, if_true] at hsafe
      simpa [parent, Position.afterPick, hlt, hmNotRemaining] using hsafe
    · intro m remaining deficit hmPos hmN hsub hholes hdeficit heven
      obtain ⟨hparentSub, hparentHoles, hmParent, hleast, hmNotRemaining⟩ :=
        insert_shifted_minimum_is_dense N m remaining hmPos hmN hsub hholes
      let parent : Position := ⟨insert m remaining, m - deficit⟩
      have hsumParent :
          parent.remaining.sum (fun x => x) = remaining.sum (fun x => x) + m := by
        simp [parent, Finset.sum_insert hmNotRemaining, Nat.add_comm]
      have hparentDeficit : parent.deficit ≤ N := by
        change m - deficit ≤ N
        omega
      have hparentCatch :
          ¬ parent.remaining.sum (fun x => x) < parent.deficit := by
        rw [hsumParent]
        change ¬ remaining.sum (fun x => x) + m < m - deficit
        omega
      have hparentEven :
          Even (parent.deficit + parent.remaining.sum (fun x => x)) := by
        rcases heven with ⟨t, ht⟩
        refine ⟨t + (m - deficit), ?_⟩
        simp only [parent, hsumParent]
        change m - deficit + (remaining.sum (fun x => x) + m) =
          (t + (m - deficit)) + (t + (m - deficit))
        omega
      have hsafe := hminimum parent (by simpa [parent] using hparentSub)
        (by simpa [parent] using hparentHoles) hparentDeficit hparentCatch hparentEven
        m (by simp [parent]) (by simpa [parent] using hleast)
      have hnotLt : ¬ m < parent.deficit := by
        change ¬ m < m - deficit
        omega
      rw [Position.moveOutcome] at hsafe
      simp only [hnotLt, if_false] at hsafe
      rw [Outcome.neg_ne_loss_iff] at hsafe
      have hsubSub : m - (m - deficit) = deficit := by omega
      simpa [parent, Position.afterPick, hnotLt, hmNotRemaining, hsubSub] using hsafe

end CatchUp
