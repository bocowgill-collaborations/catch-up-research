import catch_up.ThreeCounterSafety

/-!
# A parametric Catch-Up four-counter entry theorem

This module proves a symbolic predecessor family for the explicit three-counter endgame. The
protected player removes one of two remaining old counters, catches up, and hands the opponent a
position covered by `ThreeCounterSafety.can_avoid_loss_three_counters`.
-/

namespace CatchUpSearch.FourCounterEntry

open CatchUp

/--
A parametric family of safe four-counter positions in an `N → N + 4` game.

Let the protected player be next to move with two old counters $x<y\leq N$, two new counters
$N<a<b\leq N+4$, and deficit $d<y$. Taking $y$ ends the turn and leaves the opponent the
three-counter position with deficit $y-d$. The final hypothesis is precisely the proved safe region
for that position.
-/
theorem can_avoid_loss_four_counter_entry (N : ℕ) (owner : Player)
    (x y lower upper deficit : ℕ)
    (hxY : x < y) (hyN : y ≤ N) (hNLower : N < lower)
    (hlowerUpper : lower < upper) (hupperBound : upper ≤ N + 4)
    (hdeficitY : deficit < y)
    (hregion :
      x + (upper - lower) ≤ y - deficit ∨
        (y - deficit ≤ x ∧ x ≤ y - deficit + (upper - lower) ∧
          upper - lower ≤ x + (y - deficit) ∧ x + (y - deficit) < upper)) :
    CanAvoidLoss owner
      { remaining := {x, y, lower, upper}
        deficit
        toMove := owner } := by
  let position : GamePosition :=
    { remaining := {x, y, lower, upper}
      deficit
      toMove := owner }
  have hxLower : x < lower := by omega
  have hyLower : y < lower := by omega
  have hxUpper : x < upper := by omega
  have hyUpper : y < upper := by omega
  have heraseY : ({x, y, lower, upper} : Finset ℕ).erase y = {x, lower, upper} := by
    ext counter
    simp only [Finset.mem_erase, Finset.mem_insert, Finset.mem_singleton]
    omega
  have hterminal : ¬position.IsTerminal := by
    simp [position, GamePosition.IsTerminal, ne_of_lt hxY, ne_of_lt hxLower,
      ne_of_lt hxUpper, ne_of_lt hyLower, ne_of_lt hyUpper, ne_of_lt hlowerUpper]
    omega
  apply (can_avoid_loss_iff_exists_safe_move owner position hterminal rfl).2
  refine ⟨y, by simp [position, GamePosition.LegalMove], ?_⟩
  have hsafe := ThreeCounterSafety.can_avoid_loss_three_counters owner x lower upper
    (y - deficit) hxLower hlowerUpper (by omega) (by omega) hregion
  simpa [position, GamePosition.afterPick, Position.afterPick,
    not_lt_of_ge (Nat.le_of_lt hdeficitY), heraseY] using hsafe

end CatchUpSearch.FourCounterEntry
