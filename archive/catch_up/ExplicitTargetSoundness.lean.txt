import catch_up.FourCounterAttractor
import catch_up.ThreeCounterSafety

/-!
# Soundness of explicit Catch-Up forcing-basin targets

This module connects executable forcing-basin target predicates to solver-independent theorems
about `CatchUp.CanAvoidLoss`.
-/

namespace CatchUpSearch.FourCounterAttractor

open CatchUp

/-- Every outcome returned by the state-level top-only classifier is the exact decoded value. -/
theorem top_only_outcome?_sound (N : ℕ) (state : State) (outcome : Outcome)
    (hvalue : topOnlyOutcome? N state = some outcome) :
    (CertificateChecker.toGamePosition (N + 4) state).toPosition.value = outcome := by
  by_cases hguard : 4 ≤ N ∧ state.deficit ≤ N + 4
  · rw [topOnlyOutcome?, if_pos hguard] at hvalue
    have hsound := TopFourClassification.top_subset_value?_sound (N + 1) state.deficit
      (CertificateChecker.availableCounters (N + 4) state) outcome
      (by omega) (by omega) hvalue
    simpa [CertificateChecker.toGamePosition] using hsound
  · rw [topOnlyOutcome?, if_neg hguard] at hvalue
    contradiction

/-- Every state accepted by the symbolic top-only classifier is safe for its owner. -/
theorem can_avoid_loss_of_is_explicit_safe_top_only (N : ℕ) (owner : Player) (state : State)
    (hstate : isExplicitSafeTopOnly N owner state = true) :
    CanAvoidLoss owner (CertificateChecker.toGamePosition (N + 4) state) := by
  have hsafe : ExplicitSafeTopOnly N owner state :=
    of_decide_eq_true (by simpa [isExplicitSafeTopOnly] using hstate)
  generalize hresult : topOnlyOutcome? N state = result at hsafe
  cases result with
  | none => simp [ExplicitSafeTopOnly, hresult] at hsafe
  | some outcome =>
      have hsafety :
          if state.toMove = owner then outcome ≠ .loss else outcome ≠ .win := by
        simpa [ExplicitSafeTopOnly, hresult] using hsafe
      have hvalue := top_only_outcome?_sound N state outcome hresult
      rw [can_avoid_loss_iff_value, hvalue]
      simpa [CertificateChecker.toGamePosition] using hsafety

/-- Every decoded state in the proof-level explicit three-counter family is safe for its owner. -/
theorem can_avoid_loss_of_explicit_one_old_pair (N : ℕ) (owner : Player) (state : State)
    (hstate : ExplicitOneOldPair N owner state) :
    CanAvoidLoss owner (CertificateChecker.toGamePosition (N + 4) state) := by
  generalize hcounters : CertificateChecker.availableCounters (N + 4) state = counters at hstate
  unfold ExplicitOneOldPair at hstate
  rw [hcounters] at hstate
  cases counters with
  | nil => contradiction
  | cons x counters =>
      cases counters with
      | nil => contradiction
      | cons lower counters =>
          cases counters with
          | nil => contradiction
          | cons upper counters =>
              cases counters with
              | nil =>
                  simp only at hstate
                  rcases hstate with
                    ⟨hxN, hNLower, hlowerUpper, htoMove, hdeficitPos, hdeficitLower, hregion⟩
                  have hxLower : x < lower := by omega
                  have htoMoveEq : state.toMove = owner.other := by
                    exact ((Player.other_eq_iff_ne owner state.toMove).2 (Ne.symm htoMove)).symm
                  have hsafe := ThreeCounterSafety.can_avoid_loss_three_counters owner x lower upper
                    state.deficit hxLower hlowerUpper hdeficitPos hdeficitLower hregion
                  simpa [CertificateChecker.toGamePosition, hcounters, htoMoveEq] using hsafe
              | cons fourth counters => contradiction

/-- A successful executable explicit-target check is a proof-backed non-loss target. -/
theorem can_avoid_loss_of_is_explicit_one_old_pair (N : ℕ) (owner : Player) (state : State)
    (hstate : isExplicitOneOldPair N owner state = true) :
    CanAvoidLoss owner (CertificateChecker.toGamePosition (N + 4) state) := by
  apply can_avoid_loss_of_explicit_one_old_pair N owner state
  exact of_decide_eq_true (by simpa [isExplicitOneOldPair] using hstate)

/-- Every decoded state in the explicit four-counter entry family is safe for its owner. -/
theorem can_avoid_loss_of_explicit_four_counter_entry (N : ℕ) (owner : Player) (state : State)
    (hstate : ExplicitFourCounterEntry N owner state) :
    CanAvoidLoss owner (CertificateChecker.toGamePosition (N + 4) state) := by
  generalize hcounters : CertificateChecker.availableCounters (N + 4) state = counters at hstate
  unfold ExplicitFourCounterEntry at hstate
  rw [hcounters] at hstate
  cases counters with
  | nil => contradiction
  | cons x counters =>
      cases counters with
      | nil => contradiction
      | cons y counters =>
          cases counters with
          | nil => contradiction
          | cons lower counters =>
              cases counters with
              | nil => contradiction
              | cons upper counters =>
                  cases counters with
                  | nil =>
                      simp only at hstate
                      rcases hstate with
                        ⟨hxY, hyN, hNLower, hlowerUpper, hupperBound, htoMove,
                          hdeficitY, hregion⟩
                      have hsafe :=
                        FourCounterEntry.can_avoid_loss_four_counter_entry N owner x y lower upper
                          state.deficit hxY hyN hNLower hlowerUpper hupperBound hdeficitY hregion
                      simpa [CertificateChecker.toGamePosition, hcounters, htoMove] using hsafe
                  | cons fifth counters => contradiction

/-- A successful executable four-counter-entry check is proof-backed. -/
theorem can_avoid_loss_of_is_explicit_four_counter_entry
    (N : ℕ) (owner : Player) (state : State)
    (hstate : isExplicitFourCounterEntry N owner state = true) :
    CanAvoidLoss owner (CertificateChecker.toGamePosition (N + 4) state) := by
  apply can_avoid_loss_of_explicit_four_counter_entry N owner state
  exact of_decide_eq_true (by simpa [isExplicitFourCounterEntry] using hstate)

/-- Every node selected by the combined explicit forcing-basin target is safe for its owner. -/
theorem can_avoid_loss_of_is_explicit_target (N : ℕ) (owner : Player) (node : SolvedNode)
    (hnode : isTarget .safeTopOnlyOrOneOldPair N owner node = true) :
    CanAvoidLoss owner (CertificateChecker.toGamePosition (N + 4) node.state) := by
  have hcases : isExplicitSafeTopOnly N owner node.state = true ∨
      isExplicitOneOldPair N owner node.state = true := by
    simpa [isTarget] using hnode
  rcases hcases with htopOnly | honeOldPair
  · exact can_avoid_loss_of_is_explicit_safe_top_only N owner node.state htopOnly
  · exact can_avoid_loss_of_is_explicit_one_old_pair N owner node.state honeOldPair

/-- Every node in the enlarged explicit endgame target is safe for its owner. -/
theorem can_avoid_loss_of_is_explicit_endgame (N : ℕ) (owner : Player) (node : SolvedNode)
    (hnode : isTarget .explicitEndgame N owner node = true) :
    CanAvoidLoss owner (CertificateChecker.toGamePosition (N + 4) node.state) := by
  have hcases :
      (isExplicitSafeTopOnly N owner node.state = true ∨
        isExplicitOneOldPair N owner node.state = true) ∨
      isExplicitFourCounterEntry N owner node.state = true := by
    simpa [isTarget] using hnode
  rcases hcases with (htopOnly | honeOldPair) | hfourCounter
  · exact can_avoid_loss_of_is_explicit_safe_top_only N owner node.state htopOnly
  · exact can_avoid_loss_of_is_explicit_one_old_pair N owner node.state honeOldPair
  · exact can_avoid_loss_of_is_explicit_four_counter_entry N owner node.state hfourCounter

end CatchUpSearch.FourCounterAttractor
