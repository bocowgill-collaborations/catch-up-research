# Certified dense three-hole range

This module collects the independently checked finite certificates into range theorems.
-/

namespace CatchUpSearch.DenseThreeHoleCertifiedRange

open CatchUp

/-- The dense three-hole nonloss invariant holds at every size from nine through twenty-five. -/
@[category test, AMS 91]
theorem dense_three_hole_nonloss_nine_through_twenty_five
    (N : ℕ) (hlo : 9 ≤ N) (hi : N ≤ 25) : DenseThreeHoleNonloss N := by
  interval_cases N
  · exact DenseThreeHoleFiniteChecker.dense_three_hole_nonloss_nine
  · exact DenseThreeHoleFiniteChecker.dense_three_hole_nonloss_ten
  · exact DenseThreeHoleN11Certificate.dense_three_hole_nonloss_eleven
  · exact DenseAddTopWindowN11Certificate.dense_three_hole_nonloss_twelve
  · exact DenseThreeHoleN13PackedCertificate.dense_three_hole_nonloss_thirteen
  · exact DenseThreeHoleN14PackedCertificate.dense_three_hole_nonloss
  · exact DenseThreeHoleN15PackedCertificate.dense_three_hole_nonloss_fifteen
  · exact DenseThreeHoleN16PackedCertificate.dense_three_hole_nonloss_sixteen
  · exact DenseThreeHoleN17PackedCertificate.dense_three_hole_nonloss
  · exact DenseThreeHoleN18PackedCertificate.dense_three_hole_nonloss_eighteen
  · exact DenseThreeHoleN19PackedCertificate.dense_three_hole_nonloss_nineteen
  · exact DenseThreeHoleN20PackedCertificate.dense_three_hole_nonloss_twenty
  · exact DenseThreeHoleN21PackedCertificate.dense_three_hole_nonloss_twenty_one
  · exact DenseThreeHoleN22PackedCertificate.dense_three_hole_nonloss_twenty_two
  · exact DenseThreeHoleN23PackedCertificate.dense_three_hole_nonloss_twenty_three
  · exact DenseThreeHoleN24PackedCertificate.dense_three_hole_nonloss_twenty_four
  · exact DenseThreeHoleN25PackedCertificate.dense_three_hole_nonloss_twenty_five

/-- Compatibility form of the certified dense invariant through twenty-three. -/
@[category test, AMS 91]
theorem dense_three_hole_nonloss_nine_through_twenty_three
    (N : ℕ) (hlo : 9 ≤ N) (hi : N ≤ 23) : DenseThreeHoleNonloss N := by
  exact dense_three_hole_nonloss_nine_through_twenty_five N hlo (by omega)

/-- Compatibility form of the certified dense invariant through twenty. -/
@[category test, AMS 91]
theorem dense_three_hole_nonloss_nine_through_twenty
    (N : ℕ) (hlo : 9 ≤ N) (hi : N ≤ 20) : DenseThreeHoleNonloss N := by
  exact dense_three_hole_nonloss_nine_through_twenty_five N hlo (by omega)

/-- Every even-triangular Catch-Up interval between nine and twenty-five is a draw. -/
@[category test, AMS 91]
theorem value_Icc_eq_draw_nine_through_twenty_five_of_even_triangular
    (N : ℕ) (hlo : 9 ≤ N) (hi : N ≤ 25) (heven : Even (N * (N + 1) / 2)) :
    value (Finset.Icc 1 N) = .draw := by
  exact value_Icc_eq_draw_of_dense_three_hole_nonloss_of_even_triangular N (by omega) heven
    (dense_three_hole_nonloss_nine_through_twenty_five N hlo hi)

/-- Compatibility form of the certified conditional draw theorem through twenty-three. -/
@[category test, AMS 91]
theorem value_Icc_eq_draw_nine_through_twenty_three_of_even_triangular
    (N : ℕ) (hlo : 9 ≤ N) (hi : N ≤ 23) (heven : Even (N * (N + 1) / 2)) :
    value (Finset.Icc 1 N) = .draw := by
  exact value_Icc_eq_draw_nine_through_twenty_five_of_even_triangular N hlo (by omega) heven

/-- Compatibility form of the certified conditional draw theorem through twenty. -/
@[category test, AMS 91]
theorem value_Icc_eq_draw_nine_through_twenty_of_even_triangular
    (N : ℕ) (hlo : 9 ≤ N) (hi : N ≤ 20) (heven : Even (N * (N + 1) / 2)) :
    value (Finset.Icc 1 N) = .draw := by
  exact value_Icc_eq_draw_nine_through_twenty_five_of_even_triangular N hlo (by omega) heven

/-- Every Catch-Up interval of size at most twenty-four with even triangular total is a draw. -/
@[category test, AMS 91]
theorem value_Icc_eq_draw_le_twenty_four_of_even_triangular
    (N : ℕ) (hN : N ≤ 24) (heven : Even (N * (N + 1) / 2)) :
    value (Finset.Icc 1 N) = .draw := by
  exact CatchUpSmallBaseDraws.value_Icc_eq_draw_le_twenty_four_of_even_triangular_of_dense_range
    (fun M hlo hi => dense_three_hole_nonloss_nine_through_twenty_five M hlo (by omega))
    N hN heven

end CatchUpSearch.DenseThreeHoleCertifiedRange
