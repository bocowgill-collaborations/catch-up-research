import catch_up.CatchUpN27PackedRootDraw
import catch_up.CatchUpN28PackedRootDraw
import catch_up.DenseThreeHoleCertifiedRange

/-!
# Certified Catch-Up draws through twenty-eight counters

This module combines the dense-range reduction through twenty-four with opening-specific root
certificates at twenty-seven and twenty-eight. The intervening triangular totals are odd.
-/

namespace CatchUpSearch.CatchUpCertifiedThrough28

open CatchUp

/-- Every Catch-Up interval with at most twenty-eight counters and even total has draw value. -/
@[category test, AMS 91]
theorem value_Icc_eq_draw_le_twenty_eight_of_even_triangular
    (N : ℕ) (hN : N ≤ 28) (heven : Even (N * (N + 1) / 2)) :
    value (Finset.Icc 1 N) = .draw := by
  by_cases hsmall : N ≤ 24
  · exact DenseThreeHoleCertifiedRange.value_Icc_eq_draw_le_twenty_four_of_even_triangular
      N hsmall heven
  · have hlower : 25 ≤ N := by omega
    interval_cases N
    · norm_num [Even] at heven
    · norm_num [Even] at heven
    · exact CatchUpN27PackedRootDraw.value_Icc_one_twenty_seven_eq_draw
    · exact CatchUpN28PackedRootDraw.value_Icc_one_twenty_eight_eq_draw

end CatchUpSearch.CatchUpCertifiedThrough28
